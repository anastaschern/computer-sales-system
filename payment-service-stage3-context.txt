                payment_number,
                payment_date,

                payment_method,

                reference_number
            )

    /**
     * Регистрирует платёж в рамках существующей транзакции.
     *
     * Все проверки и INSERT выполняются на одном соединении.
     *
     * @param connection соединение с базой данных
     * @param payment данные платежа
     * @return идентификатор созданного платежа
     */
    @Override
    public Long create(Connection connection, Payment payment) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        if (payment == null) {
            throw new ValidationException(
                    "Данные платежа не должны быть null"
            );
        }

        if (payment.getPaymentId() != null) {
            throw new ValidationException(
                    "При создании платежа его идентификатор должен быть null"
            );
        }

        if (payment.getSale() == null
                || payment.getSale().getSaleId() == null) {
            throw new ValidationException(
                    "Для регистрации платежа необходимо указать продажу"
            );
        }

        payment.validatePaymentReferences();

        Long saleId = payment.getSale().getSaleId();
        validateId(saleId, "продажи");

        Long applicationId = null;
        Integer paymentNumber = null;

        String saleSql = """
                SELECT payment_type
                FROM sales
                WHERE sale_id = ?
                """;

        try (PreparedStatement statement =
                     connection.prepareStatement(saleSql)) {

            statement.setLong(1, saleId);

            try (ResultSet resultSet = statement.executeQuery()) {

                if (!resultSet.next()) {
                    throw new ValidationException(
                            "Продажа с идентификатором "
                                    + saleId + " не найдена"
                    );
                }

                String paymentType =
                        resultSet.getString("payment_type");

                if (payment.getPaymentPurpose()
                        == PaymentPurpose.INSTALLMENT_PAYMENT
                        && !"INSTALLMENT".equals(paymentType)) {

                    throw new BusinessRuleException(
                            "Платёж по графику рассрочки возможен "
                                    + "только для продажи с типом INSTALLMENT"
                    );
                }
            }
        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось проверить продажу перед регистрацией платежа",
                    exception
            );
        }

        if (payment.getApplication() != null) {

            applicationId =
                    payment.getApplication().getApplicationId();

            validateId(applicationId, "заявки на рассрочку");

            paymentNumber = payment.getPaymentNumber();

            String applicationSql = """
                    SELECT sale_id
                    FROM installment_applications
                    WHERE application_id = ?
                    """;

            try (PreparedStatement statement =
                         connection.prepareStatement(applicationSql)) {

                statement.setLong(1, applicationId);

                try (ResultSet resultSet = statement.executeQuery()) {

                    if (!resultSet.next()) {
                        throw new ValidationException(
                                "Заявка с идентификатором "
                                        + applicationId + " не найдена"
                        );
                    }

                    Long applicationSaleId =
                            resultSet.getLong("sale_id");

                    if (!Objects.equals(applicationSaleId, saleId)) {
                        throw new BusinessRuleException(
                                "Заявка на рассрочку относится "
                                        + "к другой продаже"
                        );
                    }
                }

            } catch (SQLException exception) {
                throw new DataAccessException(
                        "Не удалось проверить заявку на рассрочку",
                        exception
                );
            }

            if (payment.getPaymentPurpose()
                    == PaymentPurpose.INSTALLMENT_PAYMENT) {

                validateScheduleItem(
                        connection,
                        applicationId,
                        paymentNumber
                );
            }
        }

        try (PreparedStatement statement =
                     connection.prepareStatement(INSERT_PAYMENT_SQL)) {

            statement.setLong(1, saleId);

            if (applicationId == null) {
                statement.setNull(2, java.sql.Types.BIGINT);
                statement.setNull(3, java.sql.Types.INTEGER);
            } else {
                statement.setLong(2, applicationId);
                statement.setInt(3, paymentNumber);
            }

            statement.setTimestamp(
                    4,
                    Timestamp.valueOf(payment.getPaymentDate())
            );

            statement.setBigDecimal(5, payment.getAmount());
            statement.setString(6, payment.getPaymentMethod().name());
            statement.setString(7, payment.getPaymentPurpose().name());
            statement.setString(8, payment.getReferenceNumber());

            try (ResultSet resultSet = statement.executeQuery()) {

                if (resultSet.next()) {
                    return resultSet.getLong("payment_id");
                }

                throw new DataAccessException(
                        "База данных не вернула идентификатор платежа",
                        new SQLException(
                                "INSERT не вернул payment_id"
                        )
                );
            }

        } catch (SQLException exception) {

            if ("23503".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Нарушена связь платежа с продажей, "
                                + "заявкой или графиком рассрочки"
                );
            }

            if ("23505".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Платёж с указанными уникальными данными "
                                + "уже существует"
                );
            }

            throw new DataAccessException(
                    "Не удалось зарегистрировать платёж",
                    exception
            );
        }
    }

    /**
     * Проверяет наличие номера платежа в графике рассрочки.
     *
     * Использует соединение текущей транзакции.
     */
    private void validateScheduleItem(
            Connection connection,
            Long applicationId,
            Integer paymentNumber) {

        String sql = """
                SELECT 1
                FROM payment_schedule
                WHERE application_id = ?
                  AND payment_number = ?
                """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, applicationId);
            statement.setInt(2, paymentNumber);

            try (ResultSet resultSet = statement.executeQuery()) {

                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Указанный номер платежа отсутствует "
                                    + "в графике рассрочки"
                    );
                }
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось проверить график рассрочки",
                    exception
            );
        }
    }

    /**
     * Выполняет запрос и преобразует результат в список платежей.
     */
    private List<Payment> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        List<Payment> payments = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    payments.add(mapPayment(resultSet));
                }
            }

            return payments;

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }

    /**
     * Преобразует строку результата запроса в объект Payment.
     */
    private Payment mapPayment(ResultSet resultSet)
            throws SQLException {

        long saleId = resultSet.getLong("sale_id");

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new DataAccessException(
                        "Не найдена продажа с идентификатором "
                                + saleId,
                        new SQLException(
                                "Нарушена связь платежа с продажей"
                        )
                ));

        long applicationId =
                resultSet.getLong("application_id");

        InstallmentApplication application = null;

        if (!resultSet.wasNull()) {
            application = applicationDao.findById(applicationId)
                    .orElseThrow(() -> new DataAccessException(
                            "Не найдена заявка с идентификатором "
                                    + applicationId,
                            new SQLException(
                                    "Нарушена связь платежа с заявкой"
                            )
                    ));
        }

        int paymentNumberValue =
                resultSet.getInt("payment_number");

        Integer paymentNumber =
                resultSet.wasNull() ? null : paymentNumberValue;

        Timestamp paymentTimestamp =
                resultSet.getTimestamp("payment_date");

        Payment payment = new Payment(
                resultSet.getLong("payment_id"),
                sale,
                application,
                paymentNumber,
                paymentTimestamp.toLocalDateTime(),
                resultSet.getBigDecimal("amount"),
                PaymentMethod.valueOf(
                        resultSet.getString("payment_method")
                ),
                PaymentPurpose.valueOf(
                        resultSet.getString("payment_purpose")
                ),
                resultSet.getString("reference_number")
        );

        payment.validatePaymentReferences();

        return payment;
    }

    /**
     * Проверяет корректность идентификатора.
     */
    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор " + entityName
                            + " должен быть положительным"
            );
        }
    }
}

package by.bsuir.computersales.server.dao;

import by.bsuir.computersales.common.model.InstallmentApplication;

import by.bsuir.computersales.server.exception.DataAccessException;

import java.sql.Connection;
import java.sql.Date;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;

/**
 * JDBC-реализация доступа к графику платежей по рассрочке.
 */
public class PaymentScheduleDaoImpl implements PaymentScheduleDao {

    private final DatabaseManager databaseManager;
    private final InstallmentApplicationDao applicationDao;

    public PaymentScheduleDaoImpl(DatabaseManager databaseManager) {
        this.databaseManager = Objects.requireNonNull(
                databaseManager,
                "DatabaseManager не должен быть null"
        );

        this.applicationDao =
                new InstallmentApplicationDaoImpl(databaseManager);
    }

    /**
     * Возвращает график платежей по заявке.
     */
    @Override
    public List<PaymentScheduleItem> findByApplicationId(
            Long applicationId) {

        validateId(applicationId);

        String sql = """
                SELECT application_id,
                       payment_number,
                       due_date,
                       planned_amount
                FROM payment_schedule
                WHERE application_id = ?
                ORDER BY payment_number
                """;

        List<PaymentScheduleItem> items = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, applicationId);

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    items.add(mapScheduleItem(resultSet));
                }
            }

            return items;

        } catch (SQLException e) {
            throw new DataAccessException(
                    "Не удалось получить график платежей по заявке",
                    e
            );
        }
    }

    /**
     * Создаёт одну запись графика платежей.
     */
    @Override
    public void create(PaymentScheduleItem item) {
        validateItem(item);

        String sql = """
                INSERT INTO payment_schedule (
                    application_id,
                    payment_number,
                    due_date,
                    planned_amount
                )
                VALUES (?, ?, ?, ?)
                """;

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            fillStatement(statement, item);
            statement.executeUpdate();

        } catch (SQLException e) {
            throw new DataAccessException(
                    "Не удалось создать запись графика платежей",
                    e
            );
        }
    }


    /**
     * Создаёт несколько записей графика в одной транзакции.
     *
     * @param items список записей графика
     */
    @Override
    public void createAll(List<PaymentScheduleItem> items) {

        validateItems(items);

        if (items.isEmpty()) {
            return;
        }

        databaseManager.executeInTransaction(connection -> {
            createAll(connection, items);
            return null;
        });
    }

    /**
     * Создаёт несколько записей графика,
     * используя переданное соединение.
     *
     * Транзакцией и соединением управляет вызывающий код.
     *
     * @param connection соединение с базой данных
     * @param items список записей графика
     */
    public void createAll(
            Connection connection,
            List<PaymentScheduleItem> items) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        validateItems(items);

        if (items.isEmpty()) {
            return;
        }

        String sql = """
            INSERT INTO payment_schedule (
                application_id,
                payment_number,
                due_date,
                planned_amount
            )
            VALUES (?, ?, ?, ?)
            """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            for (PaymentScheduleItem item : items) {
                fillStatement(statement, item);
                statement.addBatch();
            }

            statement.executeBatch();

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось сохранить график платежей",
                    exception
            );
        }
    }

    /**
     * Проверяет список записей графика до начала транзакции.
     *
     * @param items список записей
     */
    private void validateItems(List<PaymentScheduleItem> items) {

        if (items == null) {
            throw new IllegalArgumentException(
                    "Список платежей не должен быть null"
            );
        }

        for (PaymentScheduleItem item : items) {
            validateItem(item);
        }
    }


    /**
     * Удаляет график платежей указанной заявки.
     */
    @Override
    public void deleteByApplicationId(Long applicationId) {
        validateId(applicationId);

        String sql = """
                DELETE FROM payment_schedule
                WHERE application_id = ?
                """;

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, applicationId);
            statement.executeUpdate();

        } catch (SQLException e) {
            throw new DataAccessException(
                    "Не удалось удалить график платежей",
                    e
            );
        }
    }

    /**
     * Преобразует строку результата SQL-запроса в объект модели.
     */
    private PaymentScheduleItem mapScheduleItem(ResultSet resultSet)
            throws SQLException {

        Long applicationId = resultSet.getLong("application_id");

        Optional<InstallmentApplication> application =
                applicationDao.findById(applicationId);

        InstallmentApplication applicationModel =
                application.orElseThrow(() ->
                        new DataAccessException(
                                "Заявка для записи графика не найдена",
                                new SQLException(
                                        "Application ID: " + applicationId
                                )
                        )
                );

        PaymentScheduleItem item = new PaymentScheduleItem();

        item.setApplication(applicationModel);
        item.setPaymentNumber(
                resultSet.getInt("payment_number")
        );
        item.setDueDate(
                resultSet.getDate("due_date").toLocalDate()
        );
        item.setPlannedAmount(
                resultSet.getBigDecimal("planned_amount")
        );

        return item;
    }

    /**
     * Заполняет параметры подготовленного SQL-запроса.
     */
    private void fillStatement(
            PreparedStatement statement,
            PaymentScheduleItem item) throws SQLException {

        statement.setLong(
                1,
                item.getApplication().getApplicationId()
        );
        statement.setInt(2, item.getPaymentNumber());
        statement.setDate(
                3,
                Date.valueOf(item.getDueDate())
        );
        statement.setBigDecimal(
                4,
                item.getPlannedAmount()
        );
    }

    /**
     * Проверяет корректность записи графика.
     */
    private void validateItem(PaymentScheduleItem item) {

        if (item == null) {
            throw new IllegalArgumentException(
                    "Запись графика платежей не должна быть null"
            );
        }

        if (item.getApplication().getApplicationId() == null
                || item.getApplication().getApplicationId() <= 0) {

            throw new IllegalArgumentException(
                    "У заявки должен быть корректный идентификатор"
            );
        }
    }

    /**
     * Проверяет идентификатор заявки.
     */
    private void validateId(Long applicationId) {
        if (applicationId == null || applicationId <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор заявки должен быть положительным"
            );
        }
    }

    /**
     * Отменяет транзакцию при возникновении ошибки.
     */
    private void rollback(Connection connection, Throwable originalError) {
        try {
            connection.rollback();
        } catch (SQLException rollbackError) {
            originalError.addSuppressed(rollbackError);
        }
    }
}

            SET stock_quantity = stock_quantity - ?
            WHERE product_id = ?
              AND is_active = TRUE
              AND stock_quantity >= ?

    /**
     * Создаёт продажу и уменьшает остатки товаров
     * в рамках одной транзакции.
     */
    @Override
    public Long createSale(Long userId, CreateSaleData saleData) {
        validateId(userId, "сотрудника");

        if (saleData == null) {
            throw new ValidationException(
                    "Данные продажи не должны быть null"
            );
        }

        List<SaleItemData> items = saleData.items();

        if (items == null || items.isEmpty()) {
            throw new ValidationException(
                    "Продажа должна содержать хотя бы один товар"
            );
        }

        // Не допускаем повторения одного товара в разных позициях.
        Set<Long> productIds = new HashSet<>();

        for (SaleItemData item : items) {
            if (item == null) {
                throw new ValidationException(
                        "Позиция продажи не должна быть null"
                );
            }

            if (!productIds.add(item.productId())) {
                throw new ValidationException(
                        "Один товар не должен повторяться "
                                + "в нескольких позициях продажи"
                );
            }
        }

        try (Connection connection = databaseManager.getConnection()) {
            connection.setAutoCommit(false);

            try {
                // Проверяем существование покупателя.
                checkCustomerExists(connection, saleData.customerId());

                // Проверяем существование и активность сотрудника.
                checkActiveUserExists(connection, userId);

                // Создаём основную запись продажи.
                Long saleId = insertSale(connection, userId, saleData);

                // Добавляем товарные позиции и уменьшаем остатки.
                int lineNumber = 1;

                for (SaleItemData item : items) {
                    insertSaleItem(
                            connection,
                            saleId,
                            lineNumber,
                            item
                    );

                    lineNumber++;
                }

                connection.commit();

                return saleId;

            } catch (SQLException exception) {
                rollback(connection, exception);

                throw new DataAccessException(
                        "Не удалось оформить продажу",
                        exception
                );

            } catch (RuntimeException exception) {
                rollback(connection, exception);
                throw exception;
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось получить соединение "
                            + "для оформления продажи",
                    exception
            );
        }
    }

    /**
     * Проверяет существование покупателя.
     */
    private void checkCustomerExists(
            Connection connection,
            Long customerId) throws SQLException {

        try (PreparedStatement statement =

                    );
                }
            }
        }
    }

    private void checkActiveUserExists(

                }
            }
        }
    }

    /**
     * Добавляет основную запись продажи.
     *
     * @return идентификатор созданной продажи
     */
    private Long insertSale(
            Connection connection,
            Long userId,
            CreateSaleData saleData) throws SQLException {

        try (PreparedStatement statement =
                     connection.prepareStatement(INSERT_SALE_SQL)) {

            statement.setLong(1, saleData.customerId());
            statement.setLong(2, userId);
            statement.setString(
                    3,
                    saleData.paymentType().name()
            );

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new SQLException(
                            "База данных не вернула идентификатор продажи"
                    );
                }

                return resultSet.getLong("sale_id");
            }
        }
    }

    /**
     * Добавляет товарную позицию продажи.
     *
     * Перед добавлением позиции остаток товара уменьшается.
     * Цена фиксируется на момент оформления продажи.
     */
    private void insertSaleItem(
            Connection connection,
            Long saleId,
            int lineNumber,
            SaleItemData item) throws SQLException {

        java.math.BigDecimal unitPrice;

        // Уменьшаем остаток только при наличии нужного количества.
        // Условие UPDATE защищает от продажи одного остатка
        // одновременно несколькими сотрудниками.
        try (PreparedStatement statement =
                     connection.prepareStatement(UPDATE_PRODUCT_STOCK_SQL)) {

            statement.setInt(1, item.quantity());
            statement.setLong(2, item.productId());
            statement.setInt(3, item.quantity());

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Товар с идентификатором "
                                    + item.productId()
                                    + " отсутствует, неактивен "
                                    + "или его остатка недостаточно"
                    );
                }

                unitPrice = resultSet.getBigDecimal("price");
            }
        }

        // Сохраняем цену из базы данных, а не от клиента.
        try (PreparedStatement statement =
                     connection.prepareStatement(INSERT_SALE_ITEM_SQL)) {

            statement.setLong(1, saleId);
            statement.setInt(2, lineNumber);
            statement.setLong(3, item.productId());
            statement.setInt(4, item.quantity());
            statement.setBigDecimal(5, unitPrice);

            statement.executeUpdate();
        }
    }

    /**
     * Отменяет транзакцию.
     * Если откат также завершился ошибкой, она добавляется
     * к исходному исключению.
     */
    private void rollback(
            Connection connection,
            Throwable originalException) {

        try {
            connection.rollback();
        } catch (SQLException rollbackException) {
            originalException.addSuppressed(rollbackException);
        }
    }

    private List<Sale> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        Map<Long, Sale> sales = new LinkedHashMap<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    Long saleId = resultSet.getLong("sale_id");

                    Sale sale = sales.computeIfAbsent(
                            saleId,
                            id -> mapSale(resultSet)
                    );

                    SaleItem item = mapSaleItem(resultSet);

                    if (item != null) {
                        sale.getItems().add(item);
                    }
                }
            }

            return new ArrayList<>(sales.values());

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }

    private Sale mapSale(ResultSet resultSet) {
        try {
            Customer customer = new Customer(
                    resultSet.getLong("customer_id"),
                    resultSet.getString("full_name"),
                    resultSet.getString("phone"),
                    resultSet.getString("email"),
                    resultSet.getString("document_reference")
            );

            Role role = new Role(
                    resultSet.getLong("role_id"),
                    RoleType.valueOf(resultSet.getString("role_name")),
                    resultSet.getString("role_description")
            );

            User user = new User(
                    resultSet.getLong("user_id"),
                    resultSet.getString("login"),
                    role,
                    resultSet.getBoolean("is_active"),
                    resultSet.getTimestamp("created_at").toLocalDateTime()
            );

            return new Sale(
                    resultSet.getLong("sale_id"),
                    customer,
                    user,
                    resultSet.getTimestamp("sale_date").toLocalDateTime(),
                    SaleStatus.valueOf(resultSet.getString("status")),
                    PaymentType.valueOf(resultSet.getString("payment_type")),
                    new ArrayList<>()
            );

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось преобразовать данные продажи",
                    exception
            );
        }
    }

    private SaleItem mapSaleItem(ResultSet resultSet)
            throws SQLException {

        long productId = resultSet.getLong("product_id");

        if (resultSet.wasNull()) {
            return null;
        }

        ProductType productType = ProductType.valueOf(
                resultSet.getString("product_type")
        );

        Category category = new Category(
                resultSet.getLong("category_id"),
                resultSet.getString("category_name"),
                resultSet.getString("category_description"),
                productType
        );

        Product product = productFactory.createProduct(
                productId,
                category,
                resultSet.getString("manufacturer"),
                resultSet.getString("model"),
                resultSet.getString("product_name"),
                resultSet.getString("specification"),
                resultSet.getBigDecimal("price"),
                resultSet.getInt("stock_quantity"),
                resultSet.getBoolean("product_is_active")
        );

        return new SaleItem(
                resultSet.getInt("line_number"),
                product,
                resultSet.getInt("quantity"),
                resultSet.getBigDecimal("unit_price")
        );
    }

    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(

                            + " должен быть положительным"
            );
        }
    }

            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"

        }

        if (status == null) {
            throw new IllegalArgumentException(
                    "Статус продажи не должен быть null"
            );
        }

        String sql = """
            UPDATE sales
            SET status = ?
            WHERE sale_id = ?
            """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setString(1, status.name());
            statement.setLong(2, saleId);

            int rowsUpdated = statement.executeUpdate();

            if (rowsUpdated == 0) {
                throw new ValidationException(
                        "Продажа с идентификатором "
                                + saleId + " не найдена"
                );
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось изменить статус продажи "
                            + saleId,
                    exception
            );
        }
    }

    public void update(

                    "Соединение с базой данных не должно быть null"
            );

                || status == InstallmentApplicationStatus.RETURNED)
                && (application.getDecisionReason() == null
                || application.getDecisionReason().isBlank())) {

            throw new ValidationException(
                    "Для отклонённой или возвращённой заявки необходимо указать причину"
            );
        }

        Long reviewerId = null;

        if (application.getReviewer() != null) {
            reviewerId = application.getReviewer().getUserId();
            validateId(reviewerId, "проверяющего сотрудника");
        }

        String sql = """
        UPDATE installment_applications
        SET reviewer_id = ?,
            decision_date = ?,
            status = ?,
            decision_reason = ?
        WHERE application_id = ?
        """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            if (reviewerId == null) {
                statement.setNull(1, java.sql.Types.BIGINT);
            } else {
                statement.setLong(1, reviewerId);
            }

            if (application.getDecisionDate() == null) {
                statement.setNull(2, java.sql.Types.TIMESTAMP);
            } else {
                statement.setTimestamp(
                        2,
                        Timestamp.valueOf(application.getDecisionDate())
                );
            }

            statement.setString(3, status.name());
            statement.setString(4, application.getDecisionReason());
            statement.setLong(5, applicationId);

                                + applicationId + " не найдена"

                );

            throw new DataAccessException(
                    "Не удалось обновить заявку на рассрочку",
                    exception
            );
        }

/**
 * Обрабатывает соединение с одним клиентом.
 *
 * Каждый экземпляр ClientHandler выполняется
 * в отдельном потоке серверного пула.
 */
public class ClientHandler implements Runnable {

    private static final long MAX_DEPTH = 20;
    private static final long MAX_REFERENCES = 1_000;
    private static final long MAX_BYTES = 1_048_576;
    private static final long MAX_ARRAY_LENGTH = 10_000;

    private final Socket clientSocket;
    private final AuthService authService;
    private final ProductDao productDao;

    private volatile boolean running = true;

    /**
     * Создаёт обработчик без сервиса авторизации.
     *
     * Используется для обратной совместимости
     * с существующими сетевыми тестами.
     */
    /**
     * Создаёт обработчик без сервисов.
     * Используется для обратной совместимости с тестами.
     */
    public ClientHandler(Socket clientSocket) {
        this(clientSocket, null, null);
    }

    /**
     * Создаёт обработчик с сервисом авторизации.
     * Используется для обратной совместимости.
     */
    public ClientHandler(Socket clientSocket, AuthService authService) {
        this(clientSocket, authService, null);
    }

    /**
     * Создаёт обработчик с сервисами авторизации и каталога товаров.
     *
     * @param clientSocket сокет клиента
     * @param authService сервис авторизации
     * @param productDao DAO для работы с товарами
     */
    public ClientHandler(
            Socket clientSocket,
            AuthService authService,
            ProductDao productDao
    ) {
        if (clientSocket == null) {
            throw new IllegalArgumentException(
                    "Сокет клиента не должен быть null"
            );
        }

        this.clientSocket = clientSocket;
        this.authService = authService;
        this.productDao = productDao;
    }
    @Override
    public void run() {
        try (
                Socket socket = clientSocket;
                ObjectOutputStream output =
                        new ObjectOutputStream(socket.getOutputStream());
                ObjectInputStream input =
                        new ObjectInputStream(socket.getInputStream())
        ) {
            input.setObjectInputFilter(createInputFilter());

            System.out.println(
                    "Начата обработка клиента: "
                            + socket.getRemoteSocketAddress()
            );

            while (running && !socket.isClosed()) {
                Object receivedObject;

                try {
                    receivedObject = input.readObject();
                } catch (EOFException | SocketException e) {
                    break;
                }

                if (!(receivedObject instanceof Request request)) {
                    break;
                }

                Response response = processRequest(request);

                output.writeObject(response);
                output.flush();
                output.reset();

                if (request.getType() == RequestType.DISCONNECT) {
                    break;
                }
            }

        } catch (InvalidClassException e) {
            System.err.println(
                    "Отклонён недопустимый объект от клиента: "
                            + e.getMessage()
            );
        } catch (IOException e) {
            System.err.println(
                    "Ошибка соединения с клиентом: "
                            + e.getMessage()
            );
        } catch (ClassNotFoundException e) {
            System.err.println(
                    "Получен неизвестный класс объекта."
            );
        } finally {
            running = false;

            System.out.println(
                    "Соединение с клиентом закрыто."
            );
        }
    }

    /**
     * Создаёт фильтр входящих сериализованных объектов.
     */
    private ObjectInputFilter createInputFilter() {
        return filterInfo -> {
            if (filterInfo.depth() > MAX_DEPTH
                    || filterInfo.references() > MAX_REFERENCES
                    || filterInfo.streamBytes() > MAX_BYTES
                    || filterInfo.arrayLength() > MAX_ARRAY_LENGTH) {
                return ObjectInputFilter.Status.REJECTED;
            }

            Class<?> serialClass = filterInfo.serialClass();

            if (serialClass == null) {
                return ObjectInputFilter.Status.UNDECIDED;
            }

            if (serialClass == Request.class
                    || serialClass == RequestType.class
                    || serialClass == String.class
                    || serialClass == java.util.UUID.class
                    || serialClass == Enum.class
                    || serialClass.isEnum()
                    || serialClass == LoginRequestData.class
                    || serialClass == CreateUserData.class
                    || serialClass == CreateSaleData.class
                    || serialClass == SaleItemData.class
                    || serialClass == PaymentType.class
                    || serialClass == Long.class
                    || serialClass == Number.class
                    || serialClass == ArrayList.class
                    || serialClass == Object[].class) {

                return ObjectInputFilter.Status.ALLOWED;
            }

            return ObjectInputFilter.Status.REJECTED;
        };
    }

    /**
     * Обрабатывает запрос клиента.
     */
    private Response processRequest(Request request) {
        if (request.getType() == RequestType.DISCONNECT) {
            running = false;

            return Response.success(
                    request.getRequestId(),
                    "Соединение завершено.",
                    null
            );
        }

        // Старый конструктор оставлен для совместимости
        // с существующими тестами ClientHandler.
        if (authService == null) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.BUSINESS_RULE,
                    "Обработка данного запроса пока не реализована."
            );
        }

        try {
            return switch (request.getType()) {
                case LOGIN -> processLogin(request);
                case LOGOUT -> processLogout(request);
                default -> processAuthenticatedRequest(request);
            };

        } catch (AuthenticationException e) {
            if (e.getErrorCode() == ErrorCode.AUTHENTICATION_REQUIRED) {
                return Response.unauthorized(
                        request.getRequestId(),
                        e.getErrorCode(),
                        e.getMessage()
                );
            }

            return Response.error(
                    request.getRequestId(),
                    e.getErrorCode(),
                    e.getMessage()
            );

        } catch (AppException e) {
            return Response.error(
                    request.getRequestId(),
                    e.getErrorCode(),
                    e.getMessage()
            );
        }
    }

    /**
     * Выполняет вход пользователя.
     */
    private Response processLogin(Request request) {
        if (!(request.getPayload() instanceof LoginRequestData loginData)) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.VALIDATION,
                    "Переданы некорректные данные для авторизации."
            );
        }

        LoginResult loginResult = authService.login(
                loginData.login(),
                loginData.password()
        );

        return Response.success(
                request.getRequestId(),
                "Авторизация выполнена успешно.",
                loginResult
        );
    }

    /**
     * Завершает пользовательскую сессию.
     */
    private Response processLogout(Request request) {
        String sessionToken = request.getSessionToken();

        if (sessionToken == null || sessionToken.isBlank()) {
            throw new AuthenticationException(
                    ErrorCode.AUTHENTICATION_REQUIRED,
                    "Для завершения сессии необходимо авторизоваться."
            );
        }

        authService.getAuthenticatedUserId(sessionToken);
        authService.logout(sessionToken);

        return Response.success(
                request.getRequestId(),
                "Сеанс пользователя завершён.",
                null
        );
    }

    /**
     * Проверяет авторизацию перед обработкой остальных запросов.
     *
     * Идентификатор пользователя извлекается из серверной сессии.
     */
    private Response processAuthenticatedRequest(Request request) {
        String sessionToken = request.getSessionToken();

        if (sessionToken == null || sessionToken.isBlank()) {
            throw new AuthenticationException(
                    ErrorCode.AUTHENTICATION_REQUIRED,
                    "Для выполнения операции необходимо авторизоваться."
            );
        }

        Long authenticatedUserId =
                authService.getAuthenticatedUserId(sessionToken);

        // На следующих этапах здесь будет подключаться
        // бизнес-логика с использованием authenticatedUserId.

        // Проверяем, что пользователь успешно авторизован.
        Objects.requireNonNull(
                authenticatedUserId,
                "Идентификатор пользователя не должен быть null"
        );

// Проверяем наличие DAO товаров.
        if (productDao == null) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.INTERNAL,
                    "Сервис каталога товаров недоступен."
            );
        }

// Обрабатываем запросы каталога.
        return switch (request.getType()) {
            case GET_PRODUCTS -> processGetProducts(request);
            case GET_PRODUCT_BY_ID -> processGetProductById(request);

            default -> Response.error(
                    request.getRequestId(),
                    ErrorCode.BUSINESS_RULE,
                    "Обработка данного запроса пока не реализована."
            );
        };
    }
    /**
     * Возвращает список активных товаров.
     *
     * @param request запрос клиента
     * @return ответ со списком товаров
     */
    private Response processGetProducts(Request request) {
        List<Product> products = productDao.findAllActive();

        return Response.success(
                request.getRequestId(),
                "Список товаров успешно получен.",
                (java.io.Serializable) new ArrayList<>(products)
        );
    }

    /**
     * Возвращает товар по его идентификатору.
     *
     * @param request запрос клиента
     * @return ответ с найденным товаром
     */
    private Response processGetProductById(Request request) {
        Object payload = request.getPayload();

        if (!(payload instanceof Long productId) || productId <= 0) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.VALIDATION,
                    "Передан некорректный идентификатор товара."
            );
        }

        return productDao.findById(productId)
                .map(product -> Response.success(
                        request.getRequestId(),
                        "Товар успешно найден.",
                        product
                ))
                .orElseGet(() -> Response.error(
                        request.getRequestId(),
                        ErrorCode.NOT_FOUND,
                        "Товар с указанным идентификатором не найден."
                ));
    }

    public PaymentPurpose getPaymentPurpose() {
        return paymentPurpose;
    }

    public void setPaymentPurpose(PaymentPurpose paymentPurpose) {
        this.paymentPurpose = Objects.requireNonNull(
                paymentPurpose,
                "Назначение платежа не должно быть null"
        );
    }

    /**
     * Проверяет согласованность назначения платежа
     * и наличия ссылки на заявку и номер платежа.
     */
    public void validatePaymentReferences() {
        boolean hasApplication = application != null;
        boolean hasPaymentNumber = paymentNumber != null;

        if (hasApplication != hasPaymentNumber) {
            throw new IllegalArgumentException(
                    "Заявка и номер платежа должны быть указаны одновременно"
            );
        }

        boolean installmentPayment =
                paymentPurpose == PaymentPurpose.INSTALLMENT_PAYMENT;

        if (installmentPayment != hasApplication) {
            throw new IllegalArgumentException(
                    "Для платежа по рассрочке необходимо указать заявку "
                            + "и номер платежа; для остальных назначений "
                            + "эти поля должны отсутствовать"
            );
        }

        if (application != null
                && application.getSale() != null
                && sale != null
                && !Objects.equals(
                application.getSale().getSaleId(),
                sale.getSaleId())) {
            throw new IllegalArgumentException(
                    "Заявка должна относиться к той же продаже, что и платёж"
            );
        }
    }
}

package by.bsuir.computersales.server.database;

import by.bsuir.computersales.server.config.ServerConfig;
import by.bsuir.computersales.server.exception.DataAccessException;
import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;

import java.sql.Connection;
import java.sql.SQLException;

/**
 * Управляет пулом подключений к базе данных
 * и транзакциями приложения.
 */
public class DatabaseManager implements AutoCloseable {

    private final HikariDataSource dataSource;

    /**
     * Операция, выполняемая внутри транзакции.
     *
     * @param <T> тип результата
     */
    @FunctionalInterface
    public interface TransactionCallback<T> {

        T execute(Connection connection) throws SQLException;
    }

    /**
     * Создаёт пул подключений на основании настроек сервера.
     *
     * @param serverConfig настройки сервера и базы данных
     */
    public DatabaseManager(ServerConfig serverConfig) {
        if (serverConfig == null) {
            throw new IllegalArgumentException(
                    "Настройки сервера не должны быть null"
            );
        }

        HikariConfig config = new HikariConfig();

        config.setJdbcUrl(serverConfig.getDatabaseUrl());
        config.setUsername(serverConfig.getDatabaseUsername());
        config.setPassword(serverConfig.getDatabasePassword());

        config.setMaximumPoolSize(serverConfig.getDatabasePoolSize());
        config.setPoolName("ComputerSalesPool");
        config.setConnectionTimeout(30000);

        this.dataSource = new HikariDataSource(config);
    }

    /**
     * Возвращает соединение с базой данных.
     *
     * @return соединение JDBC
     * @throws SQLException если получить соединение не удалось
     */
    public Connection getConnection() throws SQLException {
        return dataSource.getConnection();
    }

    /**
     * Выполняет операцию в одной транзакции.
     *
     * Если операция завершается успешно, изменения фиксируются.
     * Если возникает ошибка, все изменения отменяются.
     *
     * @param callback операция, выполняемая в транзакции
     * @param <T> тип результата
     * @return результат операции
     */
    public <T> T executeInTransaction(
            TransactionCallback<T> callback) {

        if (callback == null) {
            throw new IllegalArgumentException(
                    "Операция транзакции не должна быть null"
            );
        }

        try (Connection connection = getConnection()) {

            connection.setAutoCommit(false);

            try {
                T result = callback.execute(connection);

                connection.commit();

                return result;

            } catch (SQLException | RuntimeException exception) {

                try {
                    connection.rollback();
                } catch (SQLException rollbackException) {
                    exception.addSuppressed(rollbackException);
                }

                if (exception instanceof SQLException) {
                    throw new DataAccessException(
                            "Ошибка выполнения транзакции",
                            exception
                    );
                }

                throw (RuntimeException) exception;
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось выполнить транзакцию с базой данных",
                    exception
            );
        }
    }

    /**
     * Закрывает пул подключений.
     */
    @Override
    public void close() {
        if (dataSource != null && !dataSource.isClosed()) {
            dataSource.close();
        }
    }
}

package by.bsuir.computersales.server.network;

import by.bsuir.computersales.server.config.ServerConfig;
import by.bsuir.computersales.server.dao.UserDaoImpl;
import by.bsuir.computersales.server.database.DatabaseManager;
import by.bsuir.computersales.server.security.PasswordHasher;
import by.bsuir.computersales.server.security.SessionManager;
import by.bsuir.computersales.server.service.AuthService;
import by.bsuir.computersales.server.dao.ProductDaoImpl;
import by.bsuir.computersales.server.dao.ProductDao;

import java.io.IOException;
import java.net.InetSocketAddress;
import java.net.ServerSocket;
import java.net.Socket;
import java.nio.file.Path;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;


/**
 * Главный класс многопоточного сервера.
 *
 * Отвечает за загрузку конфигурации,
 * подключение к базе данных, авторизацию
 * пользователей и приём клиентских подключений.
 */
public class ServerMain {

    private final ServerConfig config;
    private final DatabaseManager databaseManager;
    private final ExecutorService clientPool;
    private final AuthService authService;
    private final ProductDao productDao;

    private ServerSocket serverSocket;
    private volatile boolean running;

    /**
     * Создаёт сервер и необходимые компоненты приложения.
     *
     * @param config настройки сервера
     */
    public ServerMain(ServerConfig config) {
        if (config == null) {
            throw new IllegalArgumentException(
                    "Конфигурация сервера не должна быть null"
            );
        }

        this.config = config;

        // Создаём менеджер подключений к базе данных.
        this.databaseManager = new DatabaseManager(config);

        // Создаём DAO для работы с пользователями.
        UserDaoImpl userDao = new UserDaoImpl(databaseManager);

        // Создаём DAO для работы с каталогом товаров.
        // Создаём DAO для работы с каталогом товаров.
        this.productDao = new ProductDaoImpl(databaseManager);

        // Создаём средство проверки паролей.
        PasswordHasher passwordHasher = new PasswordHasher();

        // Создаём менеджер пользовательских сессий.
        SessionManager sessionManager = new SessionManager(
                config.getSessionTimeoutSeconds()
        );

        // Создаём сервис авторизации.
        this.authService = new AuthService(
                userDao,
                passwordHasher,
                sessionManager
        );

        // Создаём пул потоков для обработки клиентов.
        this.clientPool = Executors.newFixedThreadPool(
                config.getThreadPoolSize()
        );
    }

    /**
     * Запускает сервер и начинает принимать подключения.
     */
    public void start() throws IOException {
        if (running) {
            throw new IllegalStateException(
                    "Сервер уже запущен"
            );
        }

        serverSocket = new ServerSocket();

        serverSocket.bind(new InetSocketAddress(
                config.getServerHost(),
                config.getServerPort()
        ));

        running = true;

        System.out.println("Сервер запущен.");
        System.out.println(
                "Адрес: " + config.getServerHost()
        );
        System.out.println(
                "Порт: " + config.getServerPort()
        );
        System.out.println(
                "Размер пула потоков: "
                        + config.getThreadPoolSize()
        );

        try {
            while (running) {
                Socket clientSocket = serverSocket.accept();

                System.out.println(
                        "Подключился клиент: "
                                + clientSocket.getRemoteSocketAddress()
                );

                // Передаём клиенту общий сервис авторизации.
                // Каждый ClientHandler работает в отдельном потоке.
                clientPool.execute(
                        new ClientHandler(
                                clientSocket,
                                authService,
                                productDao
                        )
                );
            }

        } catch (IOException e) {
            if (running) {
                throw e;
            }

        } finally {
            stop();
        }
    }

    /**
     * Останавливает сервер и освобождает ресурсы.
     */
    public void stop() {
        running = false;

        if (serverSocket != null && !serverSocket.isClosed()) {
            try {
                serverSocket.close();
            } catch (IOException e) {
                System.err.println(
                        "Ошибка закрытия серверного сокета: "
                                + e.getMessage()
                );
            }
        }

        clientPool.shutdown();

        try {
            if (!clientPool.awaitTermination(5, TimeUnit.SECONDS)) {
                clientPool.shutdownNow();
            }
        } catch (InterruptedException e) {
            clientPool.shutdownNow();
            Thread.currentThread().interrupt();
        }

        databaseManager.close();

        System.out.println("Сервер остановлен.");
    }

    /**
     * Точка входа в приложение.
     *
     * Первый аргумент командной строки позволяет
     * указать путь к файлу конфигурации.
     */
    public static void main(String[] args) {
        Path configPath = args.length > 0
                ? Path.of(args[0])
                : Path.of("server.properties");

        try {
            ServerConfig config = ServerConfig.load(configPath);

            ServerMain server = new ServerMain(config);

            Runtime.getRuntime().addShutdownHook(
                    new Thread(server::stop)
            );

            server.start();

        } catch (Exception e) {
            System.err.println(
                    "Не удалось запустить сервер: "
                            + e.getMessage()
            );

            e.printStackTrace();
        }
    }
}
