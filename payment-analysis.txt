package by.bsuir.computersales.common.protocol;

import java.io.Serializable;
import java.util.UUID;

/**
 * Запрос от клиента к серверу.
 */
public class Request implements Serializable {

    private static final long serialVersionUID = 1L;

    // Уникальный идентификатор запроса
    private final UUID requestId;

    // Тип операции
    private final RequestType type;

    // Токен пользовательской сессии.
    // При авторизации значение равно null.
    private final String sessionToken;

    // Данные запроса
    private final Serializable payload;

    public Request(
            RequestType type,
            String sessionToken,
            Serializable payload
    ) {
        if (type == null) {
            throw new IllegalArgumentException(
                    "Тип запроса не может быть null"
            );
        }

        this.requestId = UUID.randomUUID();
        this.type = type;
        this.sessionToken = sessionToken;
        this.payload = payload;
    }

    public UUID getRequestId() {
        return requestId;
    }

    public RequestType getType() {
        return type;
    }

    public String getSessionToken() {
        return sessionToken;
    }

    public Serializable getPayload() {
        return payload;
    }
}

package by.bsuir.computersales.common.protocol;

import java.io.Serializable;
import java.util.UUID;

/**
 * Ответ сервера клиенту.
 */
public class Response implements Serializable {

    private static final long serialVersionUID = 1L;

    private final UUID requestId;
    private final ResponseStatus status;
    private final ErrorCode errorCode;
    private final String message;
    private final Serializable payload;

    public Response(
            UUID requestId,
            ResponseStatus status,
            ErrorCode errorCode,
            String message,
            Serializable payload
    ) {
        if (requestId == null) {
            throw new IllegalArgumentException(
                    "Идентификатор запроса не может быть null"
            );
        }

        if (status == null) {
            throw new IllegalArgumentException(
                    "Статус ответа не может быть null"
            );
        }

        if (status == ResponseStatus.SUCCESS && errorCode != null) {
            throw new IllegalArgumentException(
                    "Успешный ответ не должен содержать код ошибки"
            );
        }

        if (status != ResponseStatus.SUCCESS && errorCode == null) {
            throw new IllegalArgumentException(
                    "Ответ с ошибкой должен содержать код ошибки"
            );
        }

        this.requestId = requestId;
        this.status = status;
        this.errorCode = errorCode;
        this.message = message;
        this.payload = payload;
    }

    public UUID getRequestId() {
        return requestId;
    }

    public ResponseStatus getStatus() {
        return status;
    }

    public ErrorCode getErrorCode() {
        return errorCode;
    }

    public String getMessage() {
        return message;
    }

    public Serializable getPayload() {
        return payload;
    }

    public static Response success(
            UUID requestId,
            String message,
            Serializable payload
    ) {
        return new Response(
                requestId,
                ResponseStatus.SUCCESS,
                null,
                message,
                payload
        );
    }

    public static Response error(
            UUID requestId,
            ErrorCode errorCode,
            String message
    ) {
        return new Response(
                requestId,
                ResponseStatus.ERROR,
                errorCode,
                message,
                null
        );
    }

    public static Response unauthorized(
            UUID requestId,
            ErrorCode errorCode,
            String message
    ) {
        return new Response(
                requestId,
                ResponseStatus.UNAUTHORIZED,
                errorCode,
                message,
                null
        );
    }
}

package by.bsuir.computersales.server.dao;

import by.bsuir.computersales.server.database.DatabaseManager;
import by.bsuir.computersales.common.model.InstallmentApplication;
import by.bsuir.computersales.common.model.InstallmentApplicationStatus;
import by.bsuir.computersales.common.model.Role;
import by.bsuir.computersales.common.model.RoleType;
import by.bsuir.computersales.common.model.User;
import by.bsuir.computersales.common.model.Sale;
import by.bsuir.computersales.server.exception.DataAccessException;
import by.bsuir.computersales.common.model.PaymentType;
import by.bsuir.computersales.server.exception.BusinessRuleException;
import by.bsuir.computersales.server.exception.ValidationException;

import java.sql.Statement;
import java.sql.Timestamp;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.util.ArrayList;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.List;
import java.util.Optional;
import java.util.Objects;

/**
 * JDBC-реализация доступа к данным о заявках на рассрочку.
 */
public class InstallmentApplicationDaoImpl
        implements InstallmentApplicationDao {

    private static final String SELECT_APPLICATIONS_SQL = """
        SELECT
            ia.application_id,
            ia.sale_id,
            ia.reviewer_id,
            ia.application_date,
            ia.decision_date,
            ia.term_months,
            ia.down_payment_amount,
            ia.interest_rate,
            ia.status AS application_status,
            ia.decision_reason,

            ru.user_id AS reviewer_user_id,
            ru.login AS reviewer_login,
            ru.is_active AS reviewer_is_active,
            ru.created_at AS reviewer_created_at,

            rr.role_id AS reviewer_role_id,
            rr.role_name AS reviewer_role_name,
            rr.description AS reviewer_role_description

        FROM installment_applications ia
        LEFT JOIN users ru ON ia.reviewer_id = ru.user_id
        LEFT JOIN roles rr ON ru.role_id = rr.role_id
        """;
    private final DatabaseManager databaseManager;
    private final SaleDao saleDao;

    public InstallmentApplicationDaoImpl(DatabaseManager databaseManager) {
        this.databaseManager = Objects.requireNonNull(
                databaseManager,
                "Менеджер базы данных не должен быть null"
        );

        this.saleDao = new SaleDaoImpl(databaseManager);
    }
    /**
     * Преобразует данные проверяющего сотрудника из ResultSet в User.
     *
     * @param resultSet результат SQL-запроса
     * @return проверяющий сотрудник или null, если он не назначен
     * @throws SQLException при ошибке чтения данных
     */
    private User mapReviewer(ResultSet resultSet) throws SQLException {
        long reviewerId = resultSet.getLong("reviewer_user_id");

        if (resultSet.wasNull()) {
            return null;
        }

        Role role = new Role(
                resultSet.getLong("reviewer_role_id"),
                RoleType.valueOf(
                        resultSet.getString("reviewer_role_name")
                ),
                resultSet.getString("reviewer_role_description")
        );

        return new User(
                reviewerId,
                resultSet.getString("reviewer_login"),
                role,
                resultSet.getBoolean("reviewer_is_active"),
                resultSet.getTimestamp("reviewer_created_at")
                        .toLocalDateTime()
        );
    }

    /**
     * Преобразует строку результата SQL-запроса в заявку на рассрочку.
     *
     * @param resultSet результат SQL-запроса
     * @return объект заявки
     * @throws SQLException при ошибке чтения данных
     */
    private InstallmentApplication mapApplication(ResultSet resultSet)
            throws SQLException {

        long saleId = resultSet.getLong("sale_id");

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new DataAccessException(
                        "Не найдена продажа с идентификатором " + saleId,
                        new SQLException("Нарушена связь заявки с продажей")
                ));

        java.sql.Timestamp decisionTimestamp =
                resultSet.getTimestamp("decision_date");

        return new InstallmentApplication(
                resultSet.getLong("application_id"),
                sale,
                mapReviewer(resultSet),
                resultSet.getTimestamp("application_date")
                        .toLocalDateTime(),
                decisionTimestamp == null
                        ? null
                        : decisionTimestamp.toLocalDateTime(),
                resultSet.getInt("term_months"),
                resultSet.getBigDecimal("down_payment_amount"),
                resultSet.getBigDecimal("interest_rate"),
                InstallmentApplicationStatus.valueOf(
                        resultSet.getString("application_status")
                ),
                resultSet.getString("decision_reason")
        );
    }

    /**
     * Выполняет SQL-запрос и преобразует результат в список заявок.
     *
     * @param sql SQL-запрос
     * @param parameter необязательный параметр запроса
     * @param operation описание выполняемой операции
     * @return список заявок
     */
    private List<InstallmentApplication> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        List<InstallmentApplication> applications = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    applications.add(mapApplication(resultSet));
                }
            }

            return applications;

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }
    @Override
    public List<InstallmentApplication> findAll() {
        String sql = SELECT_APPLICATIONS_SQL
                + " ORDER BY ia.application_date DESC, ia.application_id DESC";

        return executeListQuery(
                sql,
                null,
                "получить список заявок на рассрочку"
        );
    }

    @Override
    public Optional<InstallmentApplication> findById(Long applicationId) {
        validateId(applicationId, "заявки на рассрочку");

        String sql = SELECT_APPLICATIONS_SQL
                + " WHERE ia.application_id = ?";

        List<InstallmentApplication> applications = executeListQuery(
                sql,
                applicationId,
                "найти заявку на рассрочку по идентификатору"
        );

        return applications.stream().findFirst();
    }

    @Override
    public Optional<InstallmentApplication> findBySaleId(Long saleId) {
        validateId(saleId, "продажи");

        String sql = SELECT_APPLICATIONS_SQL
                + " WHERE ia.sale_id = ?";

        List<InstallmentApplication> applications = executeListQuery(
                sql,
                saleId,
                "найти заявку на рассрочку по продаже"
        );

        return applications.stream().findFirst();
    }

    @Override
    public List<InstallmentApplication> findByStatus(
            InstallmentApplicationStatus status) {

        if (status == null) {
            throw new IllegalArgumentException(
                    "Статус заявки не должен быть null"
            );
        }

        String sql = SELECT_APPLICATIONS_SQL
                + " WHERE ia.status = ?"
                + " ORDER BY ia.application_date DESC, ia.application_id DESC";

        List<InstallmentApplication> applications = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            statement.setString(1, status.name());

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    applications.add(mapApplication(resultSet));
                }
            }

            return applications;

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось получить заявки по статусу",
                    exception
            );
        }
    }

    @Override
    public List<InstallmentApplication> findByReviewerId(Long reviewerId) {
        validateId(reviewerId, "сотрудника");

        String sql = SELECT_APPLICATIONS_SQL
                + " WHERE ia.reviewer_id = ?"
                + " ORDER BY ia.application_date DESC, ia.application_id DESC";

        return executeListQuery(
                sql,
                reviewerId,
                "получить заявки сотрудника"
        );
    }

    @Override
    public Long create(InstallmentApplication application) {
        if (application == null) {
            throw new ValidationException(
                    "Заявка на рассрочку не должна быть null"
            );
        }

        if (application.getApplicationId() != null) {
            throw new ValidationException(
                    "При создании заявки её идентификатор должен быть null"
            );
        }

        Sale applicationSale = application.getSale();

        if (applicationSale == null || applicationSale.getSaleId() == null) {
            throw new ValidationException(
                    "Для создания заявки необходимо указать продажу"
            );
        }

        Long saleId = applicationSale.getSaleId();
        validateId(saleId, "продажи");

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new ValidationException(
                        "Продажа с идентификатором " + saleId + " не найдена"
                ));

        if (sale.getPaymentType() != PaymentType.INSTALLMENT) {
            throw new BusinessRuleException(
                    "Заявку на рассрочку можно создать только для продажи "
                            + "с типом оплаты INSTALLMENT"
            );
        }

        Long reviewerId = null;

        if (application.getReviewer() != null) {
            reviewerId = application.getReviewer().getUserId();
            validateId(reviewerId, "проверяющего сотрудника");
        }

        String sql = """
            INSERT INTO installment_applications (
                sale_id,
                reviewer_id,
                application_date,
                decision_date,
                term_months,
                down_payment_amount,
                interest_rate,
                status,
                decision_reason
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            RETURNING application_id
            """;

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            statement.setLong(1, saleId);

            if (reviewerId == null) {
                statement.setNull(2, java.sql.Types.BIGINT);
            } else {
                statement.setLong(2, reviewerId);
            }

            statement.setTimestamp(
                    3,
                    Timestamp.valueOf(application.getApplicationDate())
            );

            if (application.getDecisionDate() == null) {
                statement.setNull(4, java.sql.Types.TIMESTAMP);
            } else {
                statement.setTimestamp(
                        4,
                        Timestamp.valueOf(application.getDecisionDate())
                );
            }

            statement.setInt(5, application.getTermMonths());
            statement.setBigDecimal(6, application.getDownPaymentAmount());
            statement.setBigDecimal(7, application.getInterestRate());
            statement.setString(8, application.getStatus().name());
            statement.setString(9, application.getDecisionReason());

            try (ResultSet resultSet = statement.executeQuery()) {
                if (resultSet.next()) {
                    return resultSet.getLong("application_id");
                }

                throw new DataAccessException(
                        "Не удалось получить идентификатор созданной заявки",
                        new SQLException("INSERT не вернул application_id")
                );
            }

        } catch (SQLException exception) {
            if ("23505".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Для этой продажи уже существует заявка на рассрочку"
                );
            }

            throw new DataAccessException(
                    "Не удалось создать заявку на рассрочку",
                    exception
            );
        }
    }

    @Override
    public void update(InstallmentApplication application) {
        if (application == null) {
            throw new ValidationException(
                    "Заявка на рассрочку не должна быть null"
            );
        }

        try (Connection connection = databaseManager.getConnection()) {
            update(connection, application);
        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось обновить заявку на рассрочку",
                    exception
            );
        }
    }

    /**
     * Обновляет заявку, используя переданное соединение.
     *
     * Соединением и транзакцией управляет вызывающий код.
     *
     * @param connection соединение с базой данных
     * @param application обновляемая заявка
     */
    public void update(
            Connection connection,
            InstallmentApplication application) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        if (application == null) {
            throw new ValidationException(
                    "Заявка на рассрочку не должна быть null"
            );
        }

        Long applicationId = application.getApplicationId();
        validateId(applicationId, "заявки на рассрочку");

        InstallmentApplicationStatus status = application.getStatus();

        if ((status == InstallmentApplicationStatus.REJECTED
                || status == InstallmentApplicationStatus.RETURNED)
                && (application.getDecisionReason() == null
                || application.getDecisionReason().isBlank())) {

            throw new ValidationException(
                    "Для отклонённой или возвращённой заявки необходимо указать причину"
            );
        }

        Long reviewerId = null;

        if (application.getReviewer() != null) {
            reviewerId = application.getReviewer().getUserId();
            validateId(reviewerId, "проверяющего сотрудника");
        }

        String sql = """
        UPDATE installment_applications
        SET reviewer_id = ?,
            decision_date = ?,
            status = ?,
            decision_reason = ?
        WHERE application_id = ?
        """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            if (reviewerId == null) {
                statement.setNull(1, java.sql.Types.BIGINT);
            } else {
                statement.setLong(1, reviewerId);
            }

            if (application.getDecisionDate() == null) {
                statement.setNull(2, java.sql.Types.TIMESTAMP);
            } else {
                statement.setTimestamp(
                        2,
                        Timestamp.valueOf(application.getDecisionDate())
                );
            }

            statement.setString(3, status.name());
            statement.setString(4, application.getDecisionReason());
            statement.setLong(5, applicationId);

            int updatedRows = statement.executeUpdate();

            if (updatedRows == 0) {
                throw new ValidationException(
                        "Заявка с идентификатором "
                                + applicationId + " не найдена"
                );
            }

        } catch (SQLException exception) {
            if ("23503".equals(exception.getSQLState())) {
                throw new ValidationException(
                        "Указанный проверяющий сотрудник не найден"
                );
            }

            throw new DataAccessException(
                    "Не удалось обновить заявку на рассрочку",
                    exception
            );
        }
    }
    /**
     * Проверяет корректность идентификатора.
     *
     * @param id идентификатор
     * @param entityName название сущности
     */
    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор " + entityName
                            + " должен быть положительным"
            );
        }
    }
}

public class PaymentDaoImpl implements PaymentDao {

                   sale_id,
                   application_id,
                   payment_number,
                   payment_date,
                   amount,
                   payment_method,
                   payment_purpose,
                   reference_number

    private static final String INSERT_PAYMENT_SQL = """
            INSERT INTO payments (
                sale_id,
                application_id,
                payment_number,
                payment_date,
                amount,
                payment_method,
                payment_purpose,
                reference_number
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            RETURNING payment_id
            """;

    private final DatabaseManager databaseManager;
    private final SaleDao saleDao;
    private final InstallmentApplicationDao applicationDao;

    public PaymentDaoImpl(DatabaseManager databaseManager) {
        this.databaseManager = Objects.requireNonNull(
                databaseManager,
                "Менеджер базы данных не должен быть null"
        );

        this.saleDao = new SaleDaoImpl(databaseManager);
        this.applicationDao =
                new InstallmentApplicationDaoImpl(databaseManager);
    }

    @Override
    public List<Payment> findAll() {

                SELECT_PAYMENTS_SQL + " ORDER BY payment_date DESC, payment_id DESC",

                "получить список платежей"
        );
    }

    @Override
    public Optional<Payment> findById(Long paymentId) {

    }

    @Override
    public List<Payment> findBySaleId(Long saleId) {
        validateId(saleId, "продажи");

        return executeListQuery(
                SELECT_PAYMENTS_SQL
                        + " WHERE sale_id = ?"

                saleId,
                "получить платежи продажи"
        );
    }

    @Override
    public List<Payment> findByApplicationId(Long applicationId) {
        validateId(applicationId, "заявки");

        return executeListQuery(
                SELECT_PAYMENTS_SQL
                        + " WHERE application_id = ?"

    /**
     * Регистрирует новый платёж.
     */
    @Override
    public Long create(Payment payment) {
        if (payment == null) {
            throw new ValidationException(
                    "Данные платежа не должны быть null"
            );
        }

        if (payment.getPaymentId() != null) {
            throw new ValidationException(
                    "При создании платежа его идентификатор должен быть null"
            );
        }

        if (payment.getSale() == null
                || payment.getSale().getSaleId() == null) {
            throw new ValidationException(
                    "Для регистрации платежа необходимо указать продажу"
            );
        }

        payment.validatePaymentReferences();

        Long saleId = payment.getSale().getSaleId();
        validateId(saleId, "продажи");

        // Проверяем существование продажи.
        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new ValidationException(
                        "Продажа с идентификатором "
                                + saleId + " не найдена"
                ));

        Long applicationId = null;
        Integer paymentNumber = null;

        if (payment.getApplication() != null) {
            applicationId =
                    payment.getApplication().getApplicationId();

            validateId(applicationId, "заявки на рассрочку");

            Long currentApplicationId = applicationId;

            InstallmentApplication application =
                    applicationDao.findById(currentApplicationId)
                            .orElseThrow(() ->
                                    new ValidationException(
                                            "Заявка с идентификатором "
                                                    + currentApplicationId
                                                    + " не найдена"
                                    )
                            );
            if (!Objects.equals(
                    application.getSale().getSaleId(),
                    saleId)) {
                throw new BusinessRuleException(
                        "Заявка на рассрочку относится к другой продаже"
                );
            }

            paymentNumber = payment.getPaymentNumber();

            validateScheduleItem(applicationId, paymentNumber);
        }

        // Проверяем соответствие назначения платежа типу продажи.
        if (payment.getPaymentPurpose()
                == PaymentPurpose.INSTALLMENT_PAYMENT
                && sale.getPaymentType()
                != by.bsuir.computersales.common.model.PaymentType.INSTALLMENT) {
            throw new BusinessRuleException(
                    "Платёж по графику рассрочки возможен "
                            + "только для продажи с типом INSTALLMENT"
            );
        }

        String sql = INSERT_PAYMENT_SQL;

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, saleId);

            if (applicationId == null) {
                statement.setNull(2, java.sql.Types.BIGINT);
                statement.setNull(3, java.sql.Types.INTEGER);
            } else {
                statement.setLong(2, applicationId);
                statement.setInt(3, paymentNumber);
            }

            statement.setTimestamp(
                    4,
                    Timestamp.valueOf(payment.getPaymentDate())
            );

            statement.setBigDecimal(5, payment.getAmount());
            statement.setString(6, payment.getPaymentMethod().name());
            statement.setString(7, payment.getPaymentPurpose().name());
            statement.setString(8, payment.getReferenceNumber());

            try (ResultSet resultSet = statement.executeQuery()) {
                if (resultSet.next()) {
                    return resultSet.getLong("payment_id");
                }

                throw new DataAccessException(
                        "База данных не вернула идентификатор платежа",
                        new SQLException(
                                "INSERT не вернул payment_id"
                        )
                );
            }

        } catch (SQLException exception) {
            if ("23503".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Нарушена связь платежа с продажей, "
                                + "заявкой или графиком рассрочки"
                );
            }

            if ("23505".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Платёж с указанными уникальными данными "
                                + "уже существует"
                );
            }

            throw new DataAccessException(
                    "Не удалось зарегистрировать платёж",
                    exception
            );
        }
    }

    /**
     * Проверяет наличие номера платежа в графике рассрочки.
     */
    private void validateScheduleItem(
            Long applicationId,
            Integer paymentNumber) {

        String sql = """
                SELECT 1
                FROM payment_schedule
                WHERE application_id = ?
                  AND payment_number = ?
                """;

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, applicationId);
            statement.setInt(2, paymentNumber);

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Указанный номер платежа отсутствует "
                                    + "в графике рассрочки"
                    );
                }
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось проверить график рассрочки",
                    exception
            );
        }
    }

    /**
     * Выполняет запрос и преобразует результат в список платежей.
     */
    private List<Payment> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        List<Payment> payments = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    payments.add(mapPayment(resultSet));
                }
            }

            return payments;

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }

    /**
     * Преобразует строку результата запроса в объект Payment.
     */
    private Payment mapPayment(ResultSet resultSet)
            throws SQLException {

        long saleId = resultSet.getLong("sale_id");

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new DataAccessException(
                        "Не найдена продажа с идентификатором "
                                + saleId,
                        new SQLException(
                                "Нарушена связь платежа с продажей"
                        )
                ));

        long applicationId =
                resultSet.getLong("application_id");

        InstallmentApplication application = null;

        if (!resultSet.wasNull()) {
            application = applicationDao.findById(applicationId)
                    .orElseThrow(() -> new DataAccessException(
                            "Не найдена заявка с идентификатором "
                                    + applicationId,
                            new SQLException(
                                    "Нарушена связь платежа с заявкой"
                            )
                    ));
        }

        int paymentNumberValue =
                resultSet.getInt("payment_number");

        Integer paymentNumber =
                resultSet.wasNull() ? null : paymentNumberValue;

        Timestamp paymentTimestamp =
                resultSet.getTimestamp("payment_date");

        Payment payment = new Payment(
                resultSet.getLong("payment_id"),
                sale,
                application,
                paymentNumber,
                paymentTimestamp.toLocalDateTime(),
                resultSet.getBigDecimal("amount"),
                PaymentMethod.valueOf(
                        resultSet.getString("payment_method")
                ),
                PaymentPurpose.valueOf(
                        resultSet.getString("payment_purpose")
                ),
                resultSet.getString("reference_number")
        );

        payment.validatePaymentReferences();

        return payment;
    }

    /**
     * Проверяет корректность идентификатора.
     */
    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор " + entityName
                            + " должен быть положительным"
            );
        }
    }

import java.util.List;
import java.util.Objects;
import java.util.Optional;

/**

 */
public class PaymentScheduleDaoImpl implements PaymentScheduleDao {

    private final DatabaseManager databaseManager;
    private final InstallmentApplicationDao applicationDao;

    public PaymentScheduleDaoImpl(DatabaseManager databaseManager) {
        this.databaseManager = Objects.requireNonNull(
                databaseManager,
                "DatabaseManager не должен быть null"
        );

        this.applicationDao =
                new InstallmentApplicationDaoImpl(databaseManager);
    }

    /**
     * Возвращает график платежей по заявке.
     */
    @Override
    public List<PaymentScheduleItem> findByApplicationId(
            Long applicationId) {

        validateId(applicationId);

        String sql = """
                SELECT application_id,
                       payment_number,
                       due_date,
                       planned_amount
                FROM payment_schedule
                WHERE application_id = ?
                ORDER BY payment_number
                """;

        List<PaymentScheduleItem> items = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, applicationId);

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    items.add(mapScheduleItem(resultSet));
                }
            }

            return items;

        } catch (SQLException e) {
            throw new DataAccessException(
                    "Не удалось получить график платежей по заявке",
                    e
            );
        }
    }

    /**
     * Создаёт одну запись графика платежей.
     */
    @Override
    public void create(PaymentScheduleItem item) {
        validateItem(item);

        String sql = """
                INSERT INTO payment_schedule (
                    application_id,
                    payment_number,
                    due_date,
                    planned_amount
                )
                VALUES (?, ?, ?, ?)
                """;

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            fillStatement(statement, item);
            statement.executeUpdate();

        } catch (SQLException e) {
            throw new DataAccessException(
                    "Не удалось создать запись графика платежей",

            );
        }
    }

     * Создаёт несколько записей графика в одной транзакции.

     * @param items список записей графика
     */
    @Override
    public void createAll(List<PaymentScheduleItem> items) {

        validateItems(items);

        if (items.isEmpty()) {
            return;
        }

        databaseManager.executeInTransaction(connection -> {
            createAll(connection, items);
            return null;
        });
    }

    /**
     * Создаёт несколько записей графика,
     * используя переданное соединение.
     *
     * Транзакцией и соединением управляет вызывающий код.
     *
     * @param connection соединение с базой данных
     * @param items список записей графика
     */
    public void createAll(
            Connection connection,
            List<PaymentScheduleItem> items) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        validateItems(items);

        if (items.isEmpty()) {
            return;
        }

        String sql = """
            INSERT INTO payment_schedule (
                application_id,
                payment_number,
                due_date,
                planned_amount
            )
            VALUES (?, ?, ?, ?)
            """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            for (PaymentScheduleItem item : items) {
                fillStatement(statement, item);
                statement.addBatch();
            }

            statement.executeBatch();

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось сохранить график платежей",
                    exception
            );
        }
    }

    private void validateItems(List<PaymentScheduleItem> items) {

        if (items == null) {

        for (PaymentScheduleItem item : items) {

        }
    }

    private PaymentScheduleItem mapScheduleItem(ResultSet resultSet)

        Optional<InstallmentApplication> application =
                applicationDao.findById(applicationId);

        InstallmentApplication applicationModel =
                application.orElseThrow(() ->
                        new DataAccessException(
                                "Заявка для записи графика не найдена",
                                new SQLException(
                                        "Application ID: " + applicationId
                                )
                        )
                );

        PaymentScheduleItem item = new PaymentScheduleItem();

        item.setApplication(applicationModel);
        item.setPaymentNumber(
                resultSet.getInt("payment_number")
        );
        item.setDueDate(
                resultSet.getDate("due_date").toLocalDate()
        );
        item.setPlannedAmount(
                resultSet.getBigDecimal("planned_amount")
        );

        return item;
    }

    private void fillStatement(

                1,
                item.getApplication().getApplicationId()
        );
        statement.setInt(2, item.getPaymentNumber());
        statement.setDate(
                3,
                Date.valueOf(item.getDueDate())
        );
        statement.setBigDecimal(
                4,
                item.getPlannedAmount()
        );
    }

package by.bsuir.computersales.server.dao;

import by.bsuir.computersales.common.dto.SaleItemData;

import by.bsuir.computersales.server.database.DatabaseManager;

import java.sql.Connection;

/**
 * JDBC-реализация доступа к данным о продажах.
 */
public class SaleDaoImpl implements SaleDao {

    private static final String SELECT_SALES_SQL = """
            SELECT s.sale_id,
                   s.sale_date,
                   s.status,
                   s.payment_type,

                   c.customer_id,
                   c.full_name,
                   c.phone,
                   c.email,
                   c.document_reference,

                   u.user_id,
                   u.login,
                   u.is_active,
                   u.created_at,

                   r.role_id,
                   r.role_name,
                   r.description AS role_description,

                   si.line_number,
                   si.quantity,
                   si.unit_price,

                   p.product_id,
                   p.manufacturer,
                   p.model,
                   p.product_name,
                   p.specification,
                   p.price,
                   p.stock_quantity,
                   p.is_active AS product_is_active,

                   cat.category_id,
                   cat.category_name,
                   cat.description AS category_description,
                   cat.product_type

            FROM sales s
            JOIN customers c ON s.customer_id = c.customer_id
            JOIN users u ON s.user_id = u.user_id
            JOIN roles r ON u.role_id = r.role_id
            LEFT JOIN sale_items si ON s.sale_id = si.sale_id
            LEFT JOIN products p ON si.product_id = p.product_id
            LEFT JOIN categories cat ON p.category_id = cat.category_id
            """;

    private static final String FIND_ALL_SQL =
            SELECT_SALES_SQL + " ORDER BY s.sale_id, si.line_number";

    private static final String FIND_BY_ID_SQL =
            SELECT_SALES_SQL
                    + " WHERE s.sale_id = ? ORDER BY si.line_number";

    private static final String FIND_BY_CUSTOMER_ID_SQL =
            SELECT_SALES_SQL
                    + " WHERE s.customer_id = ? "
                    + "ORDER BY s.sale_date DESC, s.sale_id, si.line_number";

    private static final String FIND_BY_USER_ID_SQL =
            SELECT_SALES_SQL
                    + " WHERE s.user_id = ? "
                    + "ORDER BY s.sale_date DESC, s.sale_id, si.line_number";

    private static final String CHECK_CUSTOMER_SQL = """
            SELECT 1
            FROM customers
            WHERE customer_id = ?
            """;

    private static final String CHECK_USER_SQL = """
            SELECT 1
            FROM users
            WHERE user_id = ?
              AND is_active = TRUE
            """;

    private static final String INSERT_SALE_SQL = """
            INSERT INTO sales (customer_id, user_id, payment_type)
            VALUES (?, ?, ?)
            RETURNING sale_id
            """;

    private static final String UPDATE_PRODUCT_STOCK_SQL = """
            UPDATE products
            SET stock_quantity = stock_quantity - ?
            WHERE product_id = ?
              AND is_active = TRUE
              AND stock_quantity >= ?
            RETURNING price
            """;

    private static final String INSERT_SALE_ITEM_SQL = """
            INSERT INTO sale_items (
                sale_id,
                line_number,
                product_id,
                quantity,
                unit_price
            )
            VALUES (?, ?, ?, ?, ?)
            """;

    private final DatabaseManager databaseManager;
    private final ProductFactory productFactory;

    public SaleDaoImpl(DatabaseManager databaseManager) {
        this.databaseManager = Objects.requireNonNull(
                databaseManager,
                "Менеджер базы данных не должен быть null"
        );

        this.productFactory = new ProductFactory();
    }

    @Override
    public List<Sale> findAll() {
        return executeListQuery(
                FIND_ALL_SQL,
                null,
                "получить список продаж"
        );
    }

    @Override
    public Optional<Sale> findById(Long saleId) {
        validateId(saleId, "продажи");

        List<Sale> sales = executeListQuery(
                FIND_BY_ID_SQL,
                saleId,
                "найти продажу по идентификатору"
        );

        return sales.stream().findFirst();
    }

    @Override
    public List<Sale> findByCustomerId(Long customerId) {
        validateId(customerId, "покупателя");

        return executeListQuery(
                FIND_BY_CUSTOMER_ID_SQL,
                customerId,
                "получить продажи покупателя"
        );
    }

    @Override
    public List<Sale> findByUserId(Long userId) {
        validateId(userId, "сотрудника");

        return executeListQuery(
                FIND_BY_USER_ID_SQL,
                userId,
                "получить продажи сотрудника"
        );
    }

    /**
     * Создаёт продажу и уменьшает остатки товаров
     * в рамках одной транзакции.
     */
    @Override
    public Long createSale(Long userId, CreateSaleData saleData) {
        validateId(userId, "сотрудника");

        if (saleData == null) {
            throw new ValidationException(
                    "Данные продажи не должны быть null"
            );
        }

        List<SaleItemData> items = saleData.items();

        if (items == null || items.isEmpty()) {
            throw new ValidationException(
                    "Продажа должна содержать хотя бы один товар"
            );
        }

        // Не допускаем повторения одного товара в разных позициях.
        Set<Long> productIds = new HashSet<>();

        for (SaleItemData item : items) {
            if (item == null) {
                throw new ValidationException(
                        "Позиция продажи не должна быть null"
                );
            }

            if (!productIds.add(item.productId())) {
                throw new ValidationException(
                        "Один товар не должен повторяться "
                                + "в нескольких позициях продажи"
                );
            }
        }

        try (Connection connection = databaseManager.getConnection()) {
            connection.setAutoCommit(false);

            try {
                // Проверяем существование покупателя.
                checkCustomerExists(connection, saleData.customerId());

                // Проверяем существование и активность сотрудника.
                checkActiveUserExists(connection, userId);

                // Создаём основную запись продажи.
                Long saleId = insertSale(connection, userId, saleData);

                // Добавляем товарные позиции и уменьшаем остатки.
                int lineNumber = 1;

                for (SaleItemData item : items) {
                    insertSaleItem(
                            connection,
                            saleId,
                            lineNumber,
                            item
                    );

                    lineNumber++;
                }

                connection.commit();

                return saleId;

            } catch (SQLException exception) {
                rollback(connection, exception);

                throw new DataAccessException(
                        "Не удалось оформить продажу",
                        exception
                );

            } catch (RuntimeException exception) {
                rollback(connection, exception);
                throw exception;
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось получить соединение "
                            + "для оформления продажи",
                    exception
            );
        }
    }

    /**
     * Проверяет существование покупателя.
     */
    private void checkCustomerExists(
            Connection connection,
            Long customerId) throws SQLException {

        try (PreparedStatement statement =
                     connection.prepareStatement(CHECK_CUSTOMER_SQL)) {

            statement.setLong(1, customerId);

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Покупатель с идентификатором "
                                    + customerId + " не найден"
                    );
                }
            }
        }
    }

    /**
     * Проверяет существование и активность сотрудника.
     */
    private void checkActiveUserExists(
            Connection connection,
            Long userId) throws SQLException {

        try (PreparedStatement statement =
                     connection.prepareStatement(CHECK_USER_SQL)) {

            statement.setLong(1, userId);

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Сотрудник не найден или неактивен"
                    );
                }
            }
        }
    }

    /**
     * Добавляет основную запись продажи.
     *
     * @return идентификатор созданной продажи
     */
    private Long insertSale(
            Connection connection,
            Long userId,
            CreateSaleData saleData) throws SQLException {

        try (PreparedStatement statement =
                     connection.prepareStatement(INSERT_SALE_SQL)) {

            statement.setLong(1, saleData.customerId());
            statement.setLong(2, userId);
            statement.setString(
                    3,
                    saleData.paymentType().name()
            );

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new SQLException(
                            "База данных не вернула идентификатор продажи"
                    );
                }

                return resultSet.getLong("sale_id");
            }
        }
    }

    /**
     * Добавляет товарную позицию продажи.
     *
     * Перед добавлением позиции остаток товара уменьшается.
     * Цена фиксируется на момент оформления продажи.
     */
    private void insertSaleItem(
            Connection connection,
            Long saleId,
            int lineNumber,
            SaleItemData item) throws SQLException {

        java.math.BigDecimal unitPrice;

        // Уменьшаем остаток только при наличии нужного количества.
        // Условие UPDATE защищает от продажи одного остатка
        // одновременно несколькими сотрудниками.
        try (PreparedStatement statement =
                     connection.prepareStatement(UPDATE_PRODUCT_STOCK_SQL)) {

            statement.setInt(1, item.quantity());
            statement.setLong(2, item.productId());
            statement.setInt(3, item.quantity());

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Товар с идентификатором "
                                    + item.productId()
                                    + " отсутствует, неактивен "
                                    + "или его остатка недостаточно"
                    );
                }

                unitPrice = resultSet.getBigDecimal("price");
            }
        }

        // Сохраняем цену из базы данных, а не от клиента.
        try (PreparedStatement statement =
                     connection.prepareStatement(INSERT_SALE_ITEM_SQL)) {

            statement.setLong(1, saleId);
            statement.setInt(2, lineNumber);
            statement.setLong(3, item.productId());
            statement.setInt(4, item.quantity());
            statement.setBigDecimal(5, unitPrice);

            statement.executeUpdate();
        }
    }

    /**
     * Отменяет транзакцию.
     * Если откат также завершился ошибкой, она добавляется
     * к исходному исключению.
     */
    private void rollback(
            Connection connection,
            Throwable originalException) {

        try {
            connection.rollback();
        } catch (SQLException rollbackException) {
            originalException.addSuppressed(rollbackException);
        }
    }

    private List<Sale> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        Map<Long, Sale> sales = new LinkedHashMap<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    Long saleId = resultSet.getLong("sale_id");

                    Sale sale = sales.computeIfAbsent(
                            saleId,
                            id -> mapSale(resultSet)
                    );

                    SaleItem item = mapSaleItem(resultSet);

                    if (item != null) {
                        sale.getItems().add(item);
                    }
                }
            }

            return new ArrayList<>(sales.values());

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }

    private Sale mapSale(ResultSet resultSet) {
        try {
            Customer customer = new Customer(
                    resultSet.getLong("customer_id"),
                    resultSet.getString("full_name"),
                    resultSet.getString("phone"),
                    resultSet.getString("email"),
                    resultSet.getString("document_reference")
            );

            Role role = new Role(
                    resultSet.getLong("role_id"),
                    RoleType.valueOf(resultSet.getString("role_name")),
                    resultSet.getString("role_description")
            );

            User user = new User(
                    resultSet.getLong("user_id"),
                    resultSet.getString("login"),
                    role,
                    resultSet.getBoolean("is_active"),
                    resultSet.getTimestamp("created_at").toLocalDateTime()
            );

            return new Sale(
                    resultSet.getLong("sale_id"),
                    customer,
                    user,
                    resultSet.getTimestamp("sale_date").toLocalDateTime(),
                    SaleStatus.valueOf(resultSet.getString("status")),
                    PaymentType.valueOf(resultSet.getString("payment_type")),
                    new ArrayList<>()
            );

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось преобразовать данные продажи",
                    exception
            );
        }
    }

    private SaleItem mapSaleItem(ResultSet resultSet)
            throws SQLException {

        long productId = resultSet.getLong("product_id");

        if (resultSet.wasNull()) {
            return null;
        }

        ProductType productType = ProductType.valueOf(
                resultSet.getString("product_type")
        );

        Category category = new Category(
                resultSet.getLong("category_id"),
                resultSet.getString("category_name"),
                resultSet.getString("category_description"),
                productType
        );

        Product product = productFactory.createProduct(
                productId,
                category,
                resultSet.getString("manufacturer"),
                resultSet.getString("model"),
                resultSet.getString("product_name"),
                resultSet.getString("specification"),
                resultSet.getBigDecimal("price"),
                resultSet.getInt("stock_quantity"),
                resultSet.getBoolean("product_is_active")
        );

        return new SaleItem(
                resultSet.getInt("line_number"),
                product,
                resultSet.getInt("quantity"),
                resultSet.getBigDecimal("unit_price")
        );
    }

    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор " + entityName
                            + " должен быть положительным"
            );
        }
    }
}

package by.bsuir.computersales.server.database;

import by.bsuir.computersales.server.config.ServerConfig;
import by.bsuir.computersales.server.exception.DataAccessException;
import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;

import java.sql.Connection;
import java.sql.SQLException;

/**
 * Управляет пулом подключений к базе данных
 * и транзакциями приложения.
 */
public class DatabaseManager implements AutoCloseable {

    private final HikariDataSource dataSource;

    /**
     * Операция, выполняемая внутри транзакции.
     *
     * @param <T> тип результата
     */
    @FunctionalInterface
    public interface TransactionCallback<T> {

        T execute(Connection connection) throws SQLException;
    }

    /**
     * Создаёт пул подключений на основании настроек сервера.
     *
     * @param serverConfig настройки сервера и базы данных
     */
    public DatabaseManager(ServerConfig serverConfig) {
        if (serverConfig == null) {
            throw new IllegalArgumentException(
                    "Настройки сервера не должны быть null"
            );
        }

        HikariConfig config = new HikariConfig();

        config.setJdbcUrl(serverConfig.getDatabaseUrl());
        config.setUsername(serverConfig.getDatabaseUsername());
        config.setPassword(serverConfig.getDatabasePassword());

        config.setMaximumPoolSize(serverConfig.getDatabasePoolSize());
        config.setPoolName("ComputerSalesPool");
        config.setConnectionTimeout(30000);

        this.dataSource = new HikariDataSource(config);
    }

    /**
     * Возвращает соединение с базой данных.
     *
     * @return соединение JDBC
     * @throws SQLException если получить соединение не удалось
     */
    public Connection getConnection() throws SQLException {
        return dataSource.getConnection();
    }

    /**
     * Выполняет операцию в одной транзакции.
     *
     * Если операция завершается успешно, изменения фиксируются.
     * Если возникает ошибка, все изменения отменяются.
     *
     * @param callback операция, выполняемая в транзакции
     * @param <T> тип результата
     * @return результат операции
     */
    public <T> T executeInTransaction(
            TransactionCallback<T> callback) {

        if (callback == null) {
            throw new IllegalArgumentException(
                    "Операция транзакции не должна быть null"
            );
        }

        try (Connection connection = getConnection()) {

            connection.setAutoCommit(false);

            try {
                T result = callback.execute(connection);

                connection.commit();

                return result;

            } catch (SQLException | RuntimeException exception) {

                try {
                    connection.rollback();
                } catch (SQLException rollbackException) {
                    exception.addSuppressed(rollbackException);
                }

                if (exception instanceof SQLException) {
                    throw new DataAccessException(
                            "Ошибка выполнения транзакции",
                            exception
                    );
                }

                throw (RuntimeException) exception;
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось выполнить транзакцию с базой данных",
                    exception
            );
        }
    }

    /**
     * Закрывает пул подключений.
     */
    @Override
    public void close() {
        if (dataSource != null && !dataSource.isClosed()) {
            dataSource.close();
        }
    }
}

/**
 * Обрабатывает соединение с одним клиентом.
 *
 * Каждый экземпляр ClientHandler выполняется
 * в отдельном потоке серверного пула.
 */
public class ClientHandler implements Runnable {

    private static final long MAX_DEPTH = 20;
    private static final long MAX_REFERENCES = 1_000;
    private static final long MAX_BYTES = 1_048_576;
    private static final long MAX_ARRAY_LENGTH = 10_000;

    private final Socket clientSocket;
    private final AuthService authService;
    private final ProductDao productDao;

    private volatile boolean running = true;

    /**
     * Создаёт обработчик без сервиса авторизации.
     *
     * Используется для обратной совместимости
     * с существующими сетевыми тестами.
     */
    /**
     * Создаёт обработчик без сервисов.
     * Используется для обратной совместимости с тестами.
     */
    public ClientHandler(Socket clientSocket) {
        this(clientSocket, null, null);
    }

    /**
     * Создаёт обработчик с сервисом авторизации.
     * Используется для обратной совместимости.
     */
    public ClientHandler(Socket clientSocket, AuthService authService) {
        this(clientSocket, authService, null);
    }

    /**
     * Создаёт обработчик с сервисами авторизации и каталога товаров.
     *
     * @param clientSocket сокет клиента
     * @param authService сервис авторизации
     * @param productDao DAO для работы с товарами
     */
    public ClientHandler(
            Socket clientSocket,
            AuthService authService,
            ProductDao productDao
    ) {
        if (clientSocket == null) {
            throw new IllegalArgumentException(
                    "Сокет клиента не должен быть null"
            );
        }

        this.clientSocket = clientSocket;
        this.authService = authService;
        this.productDao = productDao;
    }
    @Override
    public void run() {
        try (
                Socket socket = clientSocket;
                ObjectOutputStream output =
                        new ObjectOutputStream(socket.getOutputStream());
                ObjectInputStream input =
                        new ObjectInputStream(socket.getInputStream())
        ) {
            input.setObjectInputFilter(createInputFilter());

            System.out.println(
                    "Начата обработка клиента: "
                            + socket.getRemoteSocketAddress()
            );

            while (running && !socket.isClosed()) {
                Object receivedObject;

                try {
                    receivedObject = input.readObject();
                } catch (EOFException | SocketException e) {
                    break;
                }

                if (!(receivedObject instanceof Request request)) {
                    break;
                }

                Response response = processRequest(request);

                output.writeObject(response);
                output.flush();
                output.reset();

                if (request.getType() == RequestType.DISCONNECT) {
                    break;
                }
            }

        } catch (InvalidClassException e) {
            System.err.println(
                    "Отклонён недопустимый объект от клиента: "
                            + e.getMessage()
            );
        } catch (IOException e) {
            System.err.println(
                    "Ошибка соединения с клиентом: "
                            + e.getMessage()
            );
        } catch (ClassNotFoundException e) {
            System.err.println(
                    "Получен неизвестный класс объекта."
            );
        } finally {
            running = false;

            System.out.println(
                    "Соединение с клиентом закрыто."
            );
        }
    }

    /**
     * Создаёт фильтр входящих сериализованных объектов.
     */
    private ObjectInputFilter createInputFilter() {
        return filterInfo -> {
            if (filterInfo.depth() > MAX_DEPTH
                    || filterInfo.references() > MAX_REFERENCES
                    || filterInfo.streamBytes() > MAX_BYTES
                    || filterInfo.arrayLength() > MAX_ARRAY_LENGTH) {
                return ObjectInputFilter.Status.REJECTED;
            }

            Class<?> serialClass = filterInfo.serialClass();

            if (serialClass == null) {
                return ObjectInputFilter.Status.UNDECIDED;
            }

            if (serialClass == Request.class
                    || serialClass == RequestType.class
                    || serialClass == String.class
                    || serialClass == java.util.UUID.class
                    || serialClass == Enum.class
                    || serialClass.isEnum()
                    || serialClass == LoginRequestData.class
                    || serialClass == CreateUserData.class
                    || serialClass == CreateSaleData.class
                    || serialClass == SaleItemData.class
                    || serialClass == PaymentType.class
                    || serialClass == Long.class
                    || serialClass == Number.class
                    || serialClass == ArrayList.class
                    || serialClass == Object[].class) {

                return ObjectInputFilter.Status.ALLOWED;
            }

            return ObjectInputFilter.Status.REJECTED;
        };
    }

    /**
     * Обрабатывает запрос клиента.
     */
    private Response processRequest(Request request) {
        if (request.getType() == RequestType.DISCONNECT) {
            running = false;

            return Response.success(
                    request.getRequestId(),
                    "Соединение завершено.",
                    null
            );
        }

        // Старый конструктор оставлен для совместимости
        // с существующими тестами ClientHandler.
        if (authService == null) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.BUSINESS_RULE,
                    "Обработка данного запроса пока не реализована."
            );
        }

        try {
            return switch (request.getType()) {
                case LOGIN -> processLogin(request);
                case LOGOUT -> processLogout(request);
                default -> processAuthenticatedRequest(request);
            };

        } catch (AuthenticationException e) {
            if (e.getErrorCode() == ErrorCode.AUTHENTICATION_REQUIRED) {
                return Response.unauthorized(
                        request.getRequestId(),
                        e.getErrorCode(),
                        e.getMessage()
                );
            }

            return Response.error(
                    request.getRequestId(),
                    e.getErrorCode(),
                    e.getMessage()
            );

        } catch (AppException e) {
            return Response.error(
                    request.getRequestId(),
                    e.getErrorCode(),
                    e.getMessage()
            );
        }
    }

    /**
     * Выполняет вход пользователя.
     */
    private Response processLogin(Request request) {
        if (!(request.getPayload() instanceof LoginRequestData loginData)) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.VALIDATION,
                    "Переданы некорректные данные для авторизации."
            );
        }

        LoginResult loginResult = authService.login(
                loginData.login(),
                loginData.password()
        );

        return Response.success(
                request.getRequestId(),
                "Авторизация выполнена успешно.",
                loginResult
        );
    }

    /**
     * Завершает пользовательскую сессию.
     */
    private Response processLogout(Request request) {
        String sessionToken = request.getSessionToken();

        if (sessionToken == null || sessionToken.isBlank()) {
            throw new AuthenticationException(
                    ErrorCode.AUTHENTICATION_REQUIRED,
                    "Для завершения сессии необходимо авторизоваться."
            );
        }

        authService.getAuthenticatedUserId(sessionToken);
        authService.logout(sessionToken);

        return Response.success(
                request.getRequestId(),
                "Сеанс пользователя завершён.",
                null
        );
    }

    /**
     * Проверяет авторизацию перед обработкой остальных запросов.
     *
     * Идентификатор пользователя извлекается из серверной сессии.
     */
    private Response processAuthenticatedRequest(Request request) {
        String sessionToken = request.getSessionToken();

        if (sessionToken == null || sessionToken.isBlank()) {
            throw new AuthenticationException(
                    ErrorCode.AUTHENTICATION_REQUIRED,
                    "Для выполнения операции необходимо авторизоваться."
            );
        }

        Long authenticatedUserId =
                authService.getAuthenticatedUserId(sessionToken);

        // На следующих этапах здесь будет подключаться
        // бизнес-логика с использованием authenticatedUserId.

        // Проверяем, что пользователь успешно авторизован.
        Objects.requireNonNull(
                authenticatedUserId,
                "Идентификатор пользователя не должен быть null"
        );

// Проверяем наличие DAO товаров.
        if (productDao == null) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.INTERNAL,
                    "Сервис каталога товаров недоступен."
            );
        }

// Обрабатываем запросы каталога.
        return switch (request.getType()) {
            case GET_PRODUCTS -> processGetProducts(request);
            case GET_PRODUCT_BY_ID -> processGetProductById(request);

            default -> Response.error(
                    request.getRequestId(),
                    ErrorCode.BUSINESS_RULE,
                    "Обработка данного запроса пока не реализована."
            );
        };
    }
    /**
     * Возвращает список активных товаров.
     *
     * @param request запрос клиента
     * @return ответ со списком товаров
     */
    private Response processGetProducts(Request request) {
        List<Product> products = productDao.findAllActive();

        return Response.success(
                request.getRequestId(),
                "Список товаров успешно получен.",
                (java.io.Serializable) new ArrayList<>(products)
        );
    }

    /**
     * Возвращает товар по его идентификатору.
     *
     * @param request запрос клиента
     * @return ответ с найденным товаром
     */
    private Response processGetProductById(Request request) {
        Object payload = request.getPayload();

        if (!(payload instanceof Long productId) || productId <= 0) {
            return Response.error(
                    request.getRequestId(),
                    ErrorCode.VALIDATION,
                    "Передан некорректный идентификатор товара."
            );
        }

        return productDao.findById(productId)
                .map(product -> Response.success(
                        request.getRequestId(),
                        "Товар успешно найден.",
                        product
                ))
                .orElseGet(() -> Response.error(
                        request.getRequestId(),
                        ErrorCode.NOT_FOUND,
                        "Товар с указанным идентификатором не найден."
                ));
    }

/**
 * Главный класс многопоточного сервера.
 *
 * Отвечает за загрузку конфигурации,
 * подключение к базе данных, авторизацию
 * пользователей и приём клиентских подключений.
 */
public class ServerMain {

    private final ServerConfig config;
    private final DatabaseManager databaseManager;
    private final ExecutorService clientPool;
    private final AuthService authService;
    private final ProductDao productDao;

    private ServerSocket serverSocket;
    private volatile boolean running;

    /**
     * Создаёт сервер и необходимые компоненты приложения.
     *
     * @param config настройки сервера
     */
    public ServerMain(ServerConfig config) {
        if (config == null) {
            throw new IllegalArgumentException(
                    "Конфигурация сервера не должна быть null"
            );
        }

        this.config = config;

        // Создаём менеджер подключений к базе данных.
        this.databaseManager = new DatabaseManager(config);

        // Создаём DAO для работы с пользователями.
        UserDaoImpl userDao = new UserDaoImpl(databaseManager);

        // Создаём DAO для работы с каталогом товаров.
        // Создаём DAO для работы с каталогом товаров.
        this.productDao = new ProductDaoImpl(databaseManager);

        // Создаём средство проверки паролей.
        PasswordHasher passwordHasher = new PasswordHasher();

        // Создаём менеджер пользовательских сессий.
        SessionManager sessionManager = new SessionManager(
                config.getSessionTimeoutSeconds()
        );

        // Создаём сервис авторизации.
        this.authService = new AuthService(
                userDao,
                passwordHasher,
                sessionManager
        );

        // Создаём пул потоков для обработки клиентов.
        this.clientPool = Executors.newFixedThreadPool(
                config.getThreadPoolSize()
        );
    }

    /**
     * Запускает сервер и начинает принимать подключения.
     */
    public void start() throws IOException {
        if (running) {
            throw new IllegalStateException(
                    "Сервер уже запущен"
            );
        }

        serverSocket = new ServerSocket();

        serverSocket.bind(new InetSocketAddress(
                config.getServerHost(),
                config.getServerPort()
        ));

        running = true;

        System.out.println("Сервер запущен.");
        System.out.println(
                "Адрес: " + config.getServerHost()
        );
        System.out.println(
                "Порт: " + config.getServerPort()
        );
        System.out.println(
                "Размер пула потоков: "
                        + config.getThreadPoolSize()
        );

        try {
            while (running) {
                Socket clientSocket = serverSocket.accept();

                System.out.println(
                        "Подключился клиент: "
                                + clientSocket.getRemoteSocketAddress()
                );

                // Передаём клиенту общий сервис авторизации.
                // Каждый ClientHandler работает в отдельном потоке.
                clientPool.execute(
                        new ClientHandler(
                                clientSocket,
                                authService,
                                productDao
                        )
                );
            }

        } catch (IOException e) {
            if (running) {
                throw e;
            }

        } finally {
            stop();
        }
    }

    /**
     * Останавливает сервер и освобождает ресурсы.
     */
    public void stop() {
        running = false;

        if (serverSocket != null && !serverSocket.isClosed()) {
            try {
                serverSocket.close();
            } catch (IOException e) {
                System.err.println(
                        "Ошибка закрытия серверного сокета: "
                                + e.getMessage()
                );
            }
        }

        clientPool.shutdown();

        try {
            if (!clientPool.awaitTermination(5, TimeUnit.SECONDS)) {
                clientPool.shutdownNow();
            }
        } catch (InterruptedException e) {
            clientPool.shutdownNow();
            Thread.currentThread().interrupt();
        }

        databaseManager.close();

        System.out.println("Сервер остановлен.");
    }

    /**
     * Точка входа в приложение.
     *
     * Первый аргумент командной строки позволяет
     * указать путь к файлу конфигурации.
     */
    public static void main(String[] args) {
        Path configPath = args.length > 0
                ? Path.of(args[0])
                : Path.of("server.properties");

        try {
            ServerConfig config = ServerConfig.load(configPath);

            ServerMain server = new ServerMain(config);

            Runtime.getRuntime().addShutdownHook(
                    new Thread(server::stop)
            );

            server.start();

        } catch (Exception e) {
            System.err.println(
                    "Не удалось запустить сервер: "
                            + e.getMessage()
            );

            e.printStackTrace();
        }
    }
}

    private List<String> validateApplicationCompleteness(

    ) {
        List<String> errors = new ArrayList<>();

        if (application == null) {
            errors.add("Заявка на рассрочку отсутствует");
            return errors;
        }

        Sale sale = application.getSale();

        if (sale == null) {
            errors.add(
                    "Не указана продажа, для которой оформляется рассрочка"
            );
            return errors;
        }

        Customer customer = sale.getCustomer();

        if (customer == null) {
            errors.add("Не указаны данные покупателя");
        } else {
            if (customer.getFullName() == null
                    || customer.getFullName().isBlank()) {
                errors.add("Не указано ФИО покупателя");
            }

            if (customer.getPhone() == null
                    || customer.getPhone().isBlank()) {

            if (customer.getDocumentReference() == null
                    || customer.getDocumentReference().isBlank()) {
                errors.add("Не указаны документные данные покупателя");
            }
        }

        if (application.getTermMonths() <= 0) {
            errors.add("Не указан срок рассрочки");
        }

        BigDecimal downPayment = application.getDownPaymentAmount();

        if (downPayment == null) {
            errors.add("Не указан первоначальный взнос");
        } else {
            if (downPayment.compareTo(BigDecimal.ZERO) < 0) {
                errors.add(
                        "Первоначальный взнос не может быть отрицательным"
                );
            }

            BigDecimal totalAmount = sale.getTotalAmount();

            if (downPayment.compareTo(totalAmount) >= 0) {
                errors.add(
                        "Первоначальный взнос должен быть меньше стоимости продажи"
                );
            }
        }

    private List<String> validateInstallmentTerm(

     *
     * @param application проверяемая заявка
     * @return список обнаруженных нарушений
     */
    private List<String> validateDownPayment(

        List<String> errors = new ArrayList<>();

        if (application == null || application.getSale() == null) {

        BigDecimal downPayment = application.getDownPaymentAmount();

        if (downPayment == null) {

            return errors;
        }

        BigDecimal totalAmount = application.getSale().getTotalAmount();
        BigDecimal minPercent = serverConfig.getMinDownPaymentPercent();

        BigDecimal minimumDownPayment = totalAmount
                .multiply(minPercent)
                .divide(new BigDecimal("100"));

        if (downPayment.compareTo(minimumDownPayment) < 0) {
            errors.add(
                    "Первоначальный взнос меньше минимально допустимого ("
                            + minPercent.stripTrailingZeros().toPlainString()

            );
        }

    }

    /**

    private List<String> validateFinancedAmount(

            return errors;
        }

        BigDecimal downPayment = application.getDownPaymentAmount();

        if (downPayment == null) {
            errors.add("Не указан первоначальный взнос");
            return errors;
        }

        BigDecimal totalAmount = application.getSale().getTotalAmount();

        BigDecimal financedAmount = totalAmount.subtract(downPayment);

        BigDecimal maxFinancedAmount =
                serverConfig.getMaxFinancedAmount();

        if (financedAmount.compareTo(maxFinancedAmount) > 0) {
            errors.add(
                    "Сумма финансирования превышает максимально допустимое значение: "
                            + maxFinancedAmount.toPlainString()
            );
        }

        return errors;
    }

    /**
     * Проверяет наличие просроченных платежей по другим рассрочкам
     * покупателя в соответствии с правилом R5.
     *
     * Просроченным считается плановый платёж, срок которого наступил
     * до текущей даты, а фактически внесённая сумма меньше плановой.
     *
     * @param application проверяемая заявка
     * @return список обнаруженных нарушений
     */
    private List<String> validateOverduePayments(
            InstallmentApplication application
    ) {
        List<String> errors = new ArrayList<>();

        if (application == null
                || application.getSale() == null
                || application.getSale().getCustomer() == null) {
            errors.add(
                    "Невозможно проверить просроченные платежи: "
                            + "отсутствуют данные заявки или покупателя"
            );
            return errors;
        }

        Long customerId = application.getSale()
                .getCustomer()
                .getCustomerId();

        if (customerId == null) {
            errors.add(
                    "Невозможно проверить просроченные платежи: "
                            + "не определён покупатель"
            );
            return errors;
        }

        LocalDate today = LocalDate.now();

        List<Sale> customerSales =
                saleDao.findByCustomerId(customerId);

        for (Sale sale : customerSales) {
            if (sale == null || sale.getSaleId() == null) {
                continue;
            }

            Optional<InstallmentApplication> otherApplicationOptional =
                    applicationDao.findBySaleId(sale.getSaleId());

            if (otherApplicationOptional.isEmpty()) {
                continue;
            }

            InstallmentApplication otherApplication =
                    otherApplicationOptional.get();

            if (Objects.equals(
                    application.getApplicationId(),
                    otherApplication.getApplicationId()
            )) {
                continue;
            }

            InstallmentApplicationStatus otherStatus =
                    otherApplication.getStatus();

            if (otherStatus != InstallmentApplicationStatus.APPROVED
                    && otherStatus
                    != InstallmentApplicationStatus.IN_PROGRESS) {
                continue;
            }

            Long otherApplicationId =
                    otherApplication.getApplicationId();

            if (otherApplicationId == null) {
                continue;
            }

            List<PaymentScheduleItem> scheduleItems =
                    paymentScheduleDao.findByApplicationId(
                            otherApplicationId
                    );

            List<Payment> actualPayments =
                    paymentDao.findByApplicationId(otherApplicationId);

            for (PaymentScheduleItem scheduleItem : scheduleItems) {
                if (scheduleItem == null
                        || scheduleItem.getDueDate() == null
                        || scheduleItem.getPlannedAmount() == null) {
                    continue;
                }

                if (!scheduleItem.getDueDate().isBefore(today)) {
                    continue;
                }

                BigDecimal paidAmount = BigDecimal.ZERO;

                for (Payment payment : actualPayments) {
                    if (payment == null
                            || !Objects.equals(
                            payment.getPaymentNumber(),
                            scheduleItem.getPaymentNumber()
                    )
                            || payment.getAmount() == null) {
                        continue;
                    }

                    paidAmount = paidAmount.add(payment.getAmount());
                }

                if (paidAmount.compareTo(
                        scheduleItem.getPlannedAmount()
                ) < 0) {
                    errors.add(
                            "У покупателя имеются просроченные платежи "
                                    + "по другой рассрочке"
                    );
                    return errors;
                }
            }
        }

        return errors;
    }

    /**
     * Проверяет количество действующих рассрочек покупателя
     * в соответствии с правилом R6.
     *
     * Активными считаются заявки со статусами APPROVED
     * и IN_PROGRESS. Текущая проверяемая заявка исключается
     * из подсчёта.
     *
     * @param application проверяемая заявка
     * @return список обнаруженных нарушений
     */
    private List<String> validateActiveInstallments(
            InstallmentApplication application
    ) {
        List<String> errors = new ArrayList<>();

        if (application == null
                || application.getSale() == null
                || application.getSale().getCustomer() == null) {
            errors.add(
                    "Невозможно проверить количество действующих рассрочек: "
                            + "отсутствуют данные заявки или покупателя"
            );
            return errors;
        }

        Long customerId = application.getSale()
                .getCustomer()
                .getCustomerId();

        if (customerId == null) {
            errors.add(
                    "Невозможно проверить количество действующих рассрочек: "
                            + "не определён покупатель"
            );
            return errors;
        }

        Long currentSaleId = application.getSale().getSaleId();

        int activeCount = 0;

        List<Sale> customerSales =
                saleDao.findByCustomerId(customerId);

        for (Sale sale : customerSales) {
            if (sale == null || sale.getSaleId() == null) {
                continue;
            }

            // Не учитываем продажу текущей заявки.
            if (Objects.equals(currentSaleId, sale.getSaleId())) {
                continue;
            }

            Optional<InstallmentApplication> otherApplicationOptional =
                    applicationDao.findBySaleId(sale.getSaleId());

            if (otherApplicationOptional.isEmpty()) {
                continue;
            }

            InstallmentApplication otherApplication =
                    otherApplicationOptional.get();

            InstallmentApplicationStatus status =
                    otherApplication.getStatus();

            if (status == InstallmentApplicationStatus.APPROVED
                    || status == InstallmentApplicationStatus.IN_PROGRESS) {
                activeCount++;
            }
        }

        int maxActiveInstallments =
                serverConfig.getMaxActiveInstallments();

        if (activeCount >= maxActiveInstallments) {
            errors.add(
                    "Превышено допустимое количество действующих рассрочек "
                            + "(максимум "
                            + maxActiveInstallments
                            + ")"
            );
        }

        return errors;
    }

    /**
     * Утверждает заявку на рассрочку и создаёт график платежей.
     *
     * Изменение статуса заявки и сохранение графика выполняются
     * в рамках одной транзакции.
     *
     * @param applicationId идентификатор заявки
     * @param reviewer сотрудник, утверждающий заявку
     * @return сформированный график платежей
     */
    public List<PaymentScheduleItem> approveApplication(
            Long applicationId,
            User reviewer
    ) {
        if (transactionExecutor == null) {
            throw new IllegalStateException(
                    "Для утверждения заявки необходим DatabaseManager"
            );
        }

        if (applicationId == null || applicationId <= 0) {
            throw new ValidationException(
                    "Идентификатор заявки должен быть положительным"
            );
        }

        if (reviewer == null
                || reviewer.getUserId() == null
                || reviewer.getUserId() <= 0) {
            throw new ValidationException(
                    "Необходимо указать сотрудника, утверждающего заявку"
            );
        }

        if (!reviewer.isActive()) {
            throw new BusinessRuleException(
                    "Неактивный сотрудник не может утверждать заявку"
            );
        }

        if (reviewer.getRole() == null
                || (reviewer.getRole().getRoleType() != RoleType.ADMIN
                && reviewer.getRole().getRoleType()
                != RoleType.INSTALLMENT_SPECIALIST)) {
            throw new BusinessRuleException(
                    "Недостаточно прав для утверждения заявки на рассрочку"
            );
        }

        InstallmentApplication application =
                applicationDao.findById(applicationId)
                        .orElseThrow(() -> new ValidationException(
                                "Заявка с идентификатором "
                                        + applicationId + " не найдена"
                        ));

        InstallmentApplicationStatus currentStatus =
                application.getStatus();

        if (currentStatus != InstallmentApplicationStatus.CREATED
                && currentStatus != InstallmentApplicationStatus.ON_REVIEW) {
            throw new BusinessRuleException(
                    "Можно утверждать только созданные заявки "
                            + "или заявки, находящиеся на рассмотрении"
            );
        }

        List<String> errors = validateApplication(application);

        if (!errors.isEmpty()) {
            throw new BusinessRuleException(
                    "Заявка не может быть утверждена: "
                            + String.join("; ", errors)
            );
        }

        LocalDateTime approvalDateTime = LocalDateTime.now();
        LocalDate approvalDate = approvalDateTime.toLocalDate();

        application.setReviewer(reviewer);
        application.setDecisionDate(approvalDateTime);
        application.setDecisionReason(null);
        application.setStatus(InstallmentApplicationStatus.APPROVED);

        ScheduleCalculator calculator = new ScheduleCalculator();

        List<PaymentScheduleItem> scheduleItems =
                calculator.calculate(application, approvalDate);

        return transactionExecutor.execute(connection -> {
            applicationDao.update(connection, application);

            paymentScheduleDao.createAll(connection, scheduleItems);

            return scheduleItems;
        });
    }

    /**
     * Выполняет комплексную проверку заявки на рассрочку
     * по правилам R1–R6.
     *
     * @param application проверяемая заявка
     * @return список обнаруженных нарушений; пустой список означает,
     * что нарушений не обнаружено
     */
    public List<String> validateApplication(
            InstallmentApplication application
    ) {
        List<String> errors = new ArrayList<>();

        // R1 — проверка полноты данных заявки.
        errors.addAll(validateApplicationCompleteness(application));

        if (application == null) {
            return errors;
        }

        // R2 — проверка срока рассрочки.
        errors.addAll(validateInstallmentTerm(application));

        // R3 и R4 — проверка первоначального взноса
        // и суммы финансирования.
        if (application.getSale() != null
                && application.getDownPaymentAmount() != null) {
            errors.addAll(validateDownPayment(application));
            errors.addAll(validateFinancedAmount(application));
        }

        // R5 — проверка просроченных платежей по другим рассрочкам.
        if (application.getSale() != null
                && application.getSale().getCustomer() != null) {
            errors.addAll(validateOverduePayments(application));
        }

        // R6 — проверка количества действующих рассрочек.
        if (application.getSale() != null
                && application.getSale().getCustomer() != null) {
            errors.addAll(validateActiveInstallments(application));
        }

        return errors;
    }
