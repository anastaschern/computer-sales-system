    private static final String SELECT_APPLICATIONS_SQL = """
        SELECT
            ia.application_id,
            ia.sale_id,
            ia.reviewer_id,
            ia.application_date,
            ia.decision_date,
            ia.term_months,
            ia.down_payment_amount,
            ia.interest_rate,
            ia.status AS application_status,
            ia.decision_reason,

            ru.user_id AS reviewer_user_id,
            ru.login AS reviewer_login,
            ru.is_active AS reviewer_is_active,
            ru.created_at AS reviewer_created_at,

            rr.role_id AS reviewer_role_id,
            rr.role_name AS reviewer_role_name,
            rr.description AS reviewer_role_description

        FROM installment_applications ia
        LEFT JOIN users ru ON ia.reviewer_id = ru.user_id
        LEFT JOIN roles rr ON ru.role_id = rr.role_id
        """;
    private final DatabaseManager databaseManager;
    private final SaleDao saleDao;

    public InstallmentApplicationDaoImpl(DatabaseManager databaseManager) {
        this.databaseManager = Objects.requireNonNull(
                databaseManager,
                "Менеджер базы данных не должен быть null"
        );

        this.saleDao = new SaleDaoImpl(databaseManager);
    }
    /**
     * Преобразует данные проверяющего сотрудника из ResultSet в User.
     *
     * @param resultSet результат SQL-запроса
     * @return проверяющий сотрудник или null, если он не назначен
     * @throws SQLException при ошибке чтения данных
     */
    private User mapReviewer(ResultSet resultSet) throws SQLException {
        long reviewerId = resultSet.getLong("reviewer_user_id");

        if (resultSet.wasNull()) {
            return null;
        }

        Role role = new Role(
                resultSet.getLong("reviewer_role_id"),
                RoleType.valueOf(
                        resultSet.getString("reviewer_role_name")
                ),
                resultSet.getString("reviewer_role_description")
        );

        return new User(
                reviewerId,
                resultSet.getString("reviewer_login"),
                role,
                resultSet.getBoolean("reviewer_is_active"),
                resultSet.getTimestamp("reviewer_created_at")
                        .toLocalDateTime()
        );
    }

    /**
     * Преобразует строку результата SQL-запроса в заявку на рассрочку.
     *
     * @param resultSet результат SQL-запроса
     * @return объект заявки
     * @throws SQLException при ошибке чтения данных
     */
    private InstallmentApplication mapApplication(ResultSet resultSet)
            throws SQLException {

        long saleId = resultSet.getLong("sale_id");

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new DataAccessException(
                        "Не найдена продажа с идентификатором " + saleId,
                        new SQLException("Нарушена связь заявки с продажей")
                ));

        java.sql.Timestamp decisionTimestamp =
                resultSet.getTimestamp("decision_date");

        return new InstallmentApplication(
                resultSet.getLong("application_id"),
                sale,
                mapReviewer(resultSet),
                resultSet.getTimestamp("application_date")
                        .toLocalDateTime(),
                decisionTimestamp == null
                        ? null
                        : decisionTimestamp.toLocalDateTime(),
                resultSet.getInt("term_months"),
                resultSet.getBigDecimal("down_payment_amount"),
                resultSet.getBigDecimal("interest_rate"),
                InstallmentApplicationStatus.valueOf(
                        resultSet.getString("application_status")
                ),
                resultSet.getString("decision_reason")
        );
    }

    /**
     * Выполняет SQL-запрос и преобразует результат в список заявок.
     *
     * @param sql SQL-запрос
     * @param parameter необязательный параметр запроса
     * @param operation описание выполняемой операции
     * @return список заявок
     */
    private List<InstallmentApplication> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        List<InstallmentApplication> applications = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    applications.add(mapApplication(resultSet));
                }
            }

            return applications;

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }
    @Override
    public List<InstallmentApplication> findAll() {
        String sql = SELECT_APPLICATIONS_SQL
                + " ORDER BY ia.application_date DESC, ia.application_id DESC";

        return executeListQuery(
                sql,
                null,
                "получить список заявок на рассрочку"
        );
    }

    @Override
    public Long create(InstallmentApplication application) {
        if (application == null) {
            throw new ValidationException(
                    "Заявка на рассрочку не должна быть null"
            );
        }

        if (application.getApplicationId() != null) {
            throw new ValidationException(
                    "При создании заявки её идентификатор должен быть null"
            );
        }

        Sale applicationSale = application.getSale();

        if (applicationSale == null || applicationSale.getSaleId() == null) {
            throw new ValidationException(
                    "Для создания заявки необходимо указать продажу"
            );
        }

        Long saleId = applicationSale.getSaleId();
        validateId(saleId, "продажи");

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new ValidationException(
                        "Продажа с идентификатором " + saleId + " не найдена"
                ));

        if (sale.getPaymentType() != PaymentType.INSTALLMENT) {
            throw new BusinessRuleException(
                    "Заявку на рассрочку можно создать только для продажи "
                            + "с типом оплаты INSTALLMENT"
            );
        }

        Long reviewerId = null;

        if (application.getReviewer() != null) {
            reviewerId = application.getReviewer().getUserId();
            validateId(reviewerId, "проверяющего сотрудника");
        }

        String sql = """
            INSERT INTO installment_applications (
                sale_id,
                reviewer_id,
                application_date,
                decision_date,
                term_months,
                down_payment_amount,
                interest_rate,
                status,
                decision_reason
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            RETURNING application_id
            """;

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            statement.setLong(1, saleId);

            if (reviewerId == null) {
                statement.setNull(2, java.sql.Types.BIGINT);
            } else {
                statement.setLong(2, reviewerId);
            }

            statement.setTimestamp(
                    3,
                    Timestamp.valueOf(application.getApplicationDate())
            );

            if (application.getDecisionDate() == null) {
                statement.setNull(4, java.sql.Types.TIMESTAMP);
            } else {
                statement.setTimestamp(
                        4,
                        Timestamp.valueOf(application.getDecisionDate())
                );
            }

            statement.setInt(5, application.getTermMonths());
            statement.setBigDecimal(6, application.getDownPaymentAmount());
            statement.setBigDecimal(7, application.getInterestRate());
            statement.setString(8, application.getStatus().name());
            statement.setString(9, application.getDecisionReason());

            try (ResultSet resultSet = statement.executeQuery()) {
                if (resultSet.next()) {
                    return resultSet.getLong("application_id");
                }

                throw new DataAccessException(
                        "Не удалось получить идентификатор созданной заявки",
                        new SQLException("INSERT не вернул application_id")
                );
            }

        } catch (SQLException exception) {
            if ("23505".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Для этой продажи уже существует заявка на рассрочку"
                );
            }

            throw new DataAccessException(
                    "Не удалось создать заявку на рассрочку",
                    exception
            );
        }
    }

    /**
     * Обновляет заявку, используя переданное соединение.
     *
     * Соединением и транзакцией управляет вызывающий код.
     *
     * @param connection соединение с базой данных
     * @param application обновляемая заявка
     */
    public void update(
            Connection connection,
            InstallmentApplication application) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        if (application == null) {
            throw new ValidationException(
                    "Заявка на рассрочку не должна быть null"
            );
        }

        Long applicationId = application.getApplicationId();
        validateId(applicationId, "заявки на рассрочку");

        InstallmentApplicationStatus status = application.getStatus();

        if ((status == InstallmentApplicationStatus.REJECTED
                || status == InstallmentApplicationStatus.RETURNED)
                && (application.getDecisionReason() == null
                || application.getDecisionReason().isBlank())) {

            throw new ValidationException(
                    "Для отклонённой или возвращённой заявки необходимо указать причину"
            );
        }

        Long reviewerId = null;

        if (application.getReviewer() != null) {
            reviewerId = application.getReviewer().getUserId();
            validateId(reviewerId, "проверяющего сотрудника");
        }

        String sql = """
        UPDATE installment_applications
        SET reviewer_id = ?,
            decision_date = ?,
            status = ?,
            decision_reason = ?
        WHERE application_id = ?
        """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            if (reviewerId == null) {
                statement.setNull(1, java.sql.Types.BIGINT);
            } else {
                statement.setLong(1, reviewerId);
            }

            if (application.getDecisionDate() == null) {
                statement.setNull(2, java.sql.Types.TIMESTAMP);
            } else {
                statement.setTimestamp(
                        2,
                        Timestamp.valueOf(application.getDecisionDate())
                );
            }

            statement.setString(3, status.name());
            statement.setString(4, application.getDecisionReason());
            statement.setLong(5, applicationId);

            int updatedRows = statement.executeUpdate();

            if (updatedRows == 0) {
                throw new ValidationException(
                        "Заявка с идентификатором "
                                + applicationId + " не найдена"
                );
            }

        } catch (SQLException exception) {
            if ("23503".equals(exception.getSQLState())) {
                throw new ValidationException(
                        "Указанный проверяющий сотрудник не найден"
                );
            }

            throw new DataAccessException(
                    "Не удалось обновить заявку на рассрочку",
                    exception
            );
        }
    }
    /**
     * Проверяет корректность идентификатора.

     * @param id идентификатор

    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор " + entityName
                            + " должен быть положительным"
            );
        }
    }

     *
     * Используется для предотвращения параллельной
     * регистрации первоначальных взносов и платежей по графику.
     *
     * @param connection соединение текущей транзакции
     * @param applicationId идентификатор заявки
     */
    @Override
    public void lockById(Connection connection, Long applicationId) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        validateId(applicationId, "заявки на рассрочку");

        String sql = """
            SELECT application_id
            FROM installment_applications
            WHERE application_id = ?
            FOR UPDATE
            """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, applicationId);

            try (ResultSet resultSet = statement.executeQuery()) {

                if (!resultSet.next()) {
                    throw new ValidationException(
                            "Заявка на рассрочку с идентификатором "
                                    + applicationId + " не найдена"
                    );
                }
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось заблокировать заявку на рассрочку "
                            + applicationId,
                    exception
            );
        }
    }

    /**
     * Регистрирует платёж в рамках существующей транзакции.
     *
     * Все проверки и INSERT выполняются на одном соединении.
     *

     * @param payment данные платежа
     * @return идентификатор созданного платежа
     */
    @Override
    public Long create(Connection connection, Payment payment) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        if (payment == null) {
            throw new ValidationException(
                    "Данные платежа не должны быть null"
            );
        }

        if (payment.getPaymentId() != null) {
            throw new ValidationException(
                    "При создании платежа его идентификатор должен быть null"
            );
        }

        if (payment.getSale() == null
                || payment.getSale().getSaleId() == null) {
            throw new ValidationException(
                    "Для регистрации платежа необходимо указать продажу"
            );
        }

        payment.validatePaymentReferences();

        Long saleId = payment.getSale().getSaleId();
        validateId(saleId, "продажи");

        Long applicationId = null;
        Integer paymentNumber = null;

        String saleSql = """
                SELECT payment_type
                FROM sales
                WHERE sale_id = ?
                """;

        try (PreparedStatement statement =
                     connection.prepareStatement(saleSql)) {

            statement.setLong(1, saleId);

            try (ResultSet resultSet = statement.executeQuery()) {

                if (!resultSet.next()) {
                    throw new ValidationException(
                            "Продажа с идентификатором "
                                    + saleId + " не найдена"
                    );
                }

                String paymentType =
                        resultSet.getString("payment_type");

                if (payment.getPaymentPurpose()
                        == PaymentPurpose.INSTALLMENT_PAYMENT
                        && !"INSTALLMENT".equals(paymentType)) {

                    throw new BusinessRuleException(
                            "Платёж по графику рассрочки возможен "
                                    + "только для продажи с типом INSTALLMENT"
                    );
                }
            }
        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось проверить продажу перед регистрацией платежа",
                    exception
            );
        }

        if (payment.getApplication() != null) {

            applicationId =
                    payment.getApplication().getApplicationId();

            validateId(applicationId, "заявки на рассрочку");

            paymentNumber = payment.getPaymentNumber();

            String applicationSql = """
                    SELECT sale_id
                    FROM installment_applications
                    WHERE application_id = ?
                    """;

            try (PreparedStatement statement =
                         connection.prepareStatement(applicationSql)) {

                statement.setLong(1, applicationId);

                try (ResultSet resultSet = statement.executeQuery()) {

                    if (!resultSet.next()) {
                        throw new ValidationException(
                                "Заявка с идентификатором "
                                        + applicationId + " не найдена"
                        );
                    }

                    Long applicationSaleId =
                            resultSet.getLong("sale_id");

                    if (!Objects.equals(applicationSaleId, saleId)) {
                        throw new BusinessRuleException(
                                "Заявка на рассрочку относится "
                                        + "к другой продаже"
                        );
                    }
                }

            } catch (SQLException exception) {
                throw new DataAccessException(
                        "Не удалось проверить заявку на рассрочку",
                        exception
                );
            }

            if (payment.getPaymentPurpose()
                    == PaymentPurpose.INSTALLMENT_PAYMENT) {

                validateScheduleItem(
                        connection,
                        applicationId,
                        paymentNumber
                );
            }
        }

        try (PreparedStatement statement =
                     connection.prepareStatement(INSERT_PAYMENT_SQL)) {

            statement.setLong(1, saleId);

            if (applicationId == null) {
                statement.setNull(2, java.sql.Types.BIGINT);
                statement.setNull(3, java.sql.Types.INTEGER);
            } else {
                statement.setLong(2, applicationId);
                statement.setInt(3, paymentNumber);
            }

            statement.setTimestamp(
                    4,
                    Timestamp.valueOf(payment.getPaymentDate())
            );

            statement.setBigDecimal(5, payment.getAmount());
            statement.setString(6, payment.getPaymentMethod().name());
            statement.setString(7, payment.getPaymentPurpose().name());
            statement.setString(8, payment.getReferenceNumber());

            try (ResultSet resultSet = statement.executeQuery()) {

                if (resultSet.next()) {
                    return resultSet.getLong("payment_id");
                }

                throw new DataAccessException(
                        "База данных не вернула идентификатор платежа",
                        new SQLException(
                                "INSERT не вернул payment_id"
                        )
                );
            }

        } catch (SQLException exception) {

            if ("23503".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Нарушена связь платежа с продажей, "
                                + "заявкой или графиком рассрочки"
                );
            }

            if ("23505".equals(exception.getSQLState())) {
                throw new BusinessRuleException(
                        "Платёж с указанными уникальными данными "
                                + "уже существует"
                );
            }

            throw new DataAccessException(
                    "Не удалось зарегистрировать платёж",
                    exception
            );
        }
    }

    /**
     * Проверяет наличие номера платежа в графике рассрочки.
     *
     * Использует соединение текущей транзакции.
     */
    private void validateScheduleItem(
            Connection connection,
            Long applicationId,
            Integer paymentNumber) {

        String sql = """
                SELECT 1
                FROM payment_schedule
                WHERE application_id = ?
                  AND payment_number = ?
                """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, applicationId);
            statement.setInt(2, paymentNumber);

            try (ResultSet resultSet = statement.executeQuery()) {

                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Указанный номер платежа отсутствует "
                                    + "в графике рассрочки"
                    );
                }
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось проверить график рассрочки",
                    exception
            );
        }
    }

    /**
     * Выполняет запрос и преобразует результат в список платежей.
     */
    private List<Payment> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        List<Payment> payments = new ArrayList<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    payments.add(mapPayment(resultSet));
                }
            }

            return payments;

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }

    /**
     * Преобразует строку результата запроса в объект Payment.
     */
    private Payment mapPayment(ResultSet resultSet)
            throws SQLException {

        long saleId = resultSet.getLong("sale_id");

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new DataAccessException(
                        "Не найдена продажа с идентификатором "
                                + saleId,
                        new SQLException(
                                "Нарушена связь платежа с продажей"
                        )
                ));

        long applicationId =
                resultSet.getLong("application_id");

        InstallmentApplication application = null;

        if (!resultSet.wasNull()) {
            application = applicationDao.findById(applicationId)
                    .orElseThrow(() -> new DataAccessException(
                            "Не найдена заявка с идентификатором "
                                    + applicationId,
                            new SQLException(
                                    "Нарушена связь платежа с заявкой"
                            )
                    ));
        }

        int paymentNumberValue =
                resultSet.getInt("payment_number");

        Integer paymentNumber =
                resultSet.wasNull() ? null : paymentNumberValue;

        Timestamp paymentTimestamp =
                resultSet.getTimestamp("payment_date");

        Payment payment = new Payment(
                resultSet.getLong("payment_id"),
                sale,
                application,
                paymentNumber,
                paymentTimestamp.toLocalDateTime(),
                resultSet.getBigDecimal("amount"),
                PaymentMethod.valueOf(
                        resultSet.getString("payment_method")
                ),
                PaymentPurpose.valueOf(
                        resultSet.getString("payment_purpose")
                ),
                resultSet.getString("reference_number")
        );

        payment.validatePaymentReferences();

        return payment;
    }

    /**
     * Проверяет корректность идентификатора.
     */
    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор " + entityName
                            + " должен быть положительным"
            );
        }
    }

    public void createAll(

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );

        if (items.isEmpty()) {

            INSERT INTO payment_schedule (
                application_id,
                payment_number,
                due_date,
                planned_amount

                     connection.prepareStatement(sql)) {

    /**
     * Создаёт продажу и уменьшает остатки товаров
     * в рамках одной транзакции.
     */
    @Override
    public Long createSale(Long userId, CreateSaleData saleData) {
        validateId(userId, "сотрудника");

        if (saleData == null) {
            throw new ValidationException(
                    "Данные продажи не должны быть null"
            );
        }

        List<SaleItemData> items = saleData.items();

        if (items == null || items.isEmpty()) {
            throw new ValidationException(
                    "Продажа должна содержать хотя бы один товар"
            );
        }

        // Не допускаем повторения одного товара в разных позициях.
        Set<Long> productIds = new HashSet<>();

        for (SaleItemData item : items) {
            if (item == null) {
                throw new ValidationException(
                        "Позиция продажи не должна быть null"
                );
            }

            if (!productIds.add(item.productId())) {
                throw new ValidationException(
                        "Один товар не должен повторяться "
                                + "в нескольких позициях продажи"
                );
            }
        }

        try (Connection connection = databaseManager.getConnection()) {
            connection.setAutoCommit(false);

            try {
                // Проверяем существование покупателя.
                checkCustomerExists(connection, saleData.customerId());

                // Проверяем существование и активность сотрудника.
                checkActiveUserExists(connection, userId);

                // Создаём основную запись продажи.
                Long saleId = insertSale(connection, userId, saleData);

                // Добавляем товарные позиции и уменьшаем остатки.
                int lineNumber = 1;

                for (SaleItemData item : items) {
                    insertSaleItem(
                            connection,
                            saleId,
                            lineNumber,
                            item
                    );

                    lineNumber++;
                }

                connection.commit();

                return saleId;

            } catch (SQLException exception) {
                rollback(connection, exception);

                throw new DataAccessException(
                        "Не удалось оформить продажу",
                        exception
                );

            } catch (RuntimeException exception) {
                rollback(connection, exception);
                throw exception;
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось получить соединение "
                            + "для оформления продажи",
                    exception
            );
        }
    }

    private Long insertSale(
            Connection connection,

                     connection.prepareStatement(INSERT_SALE_SQL)) {

            statement.setLong(1, saleData.customerId());
            statement.setLong(2, userId);

                    3,

            );

    /**
     * Добавляет товарную позицию продажи.
     *
     * Перед добавлением позиции остаток товара уменьшается.
     * Цена фиксируется на момент оформления продажи.
     */
    private void insertSaleItem(
            Connection connection,
            Long saleId,
            int lineNumber,
            SaleItemData item) throws SQLException {

        java.math.BigDecimal unitPrice;

        // Уменьшаем остаток только при наличии нужного количества.
        // Условие UPDATE защищает от продажи одного остатка
        // одновременно несколькими сотрудниками.
        try (PreparedStatement statement =
                     connection.prepareStatement(UPDATE_PRODUCT_STOCK_SQL)) {

            statement.setInt(1, item.quantity());
            statement.setLong(2, item.productId());
            statement.setInt(3, item.quantity());

            try (ResultSet resultSet = statement.executeQuery()) {
                if (!resultSet.next()) {
                    throw new BusinessRuleException(
                            "Товар с идентификатором "
                                    + item.productId()
                                    + " отсутствует, неактивен "
                                    + "или его остатка недостаточно"
                    );
                }

                unitPrice = resultSet.getBigDecimal("price");
            }
        }

        // Сохраняем цену из базы данных, а не от клиента.
        try (PreparedStatement statement =
                     connection.prepareStatement(INSERT_SALE_ITEM_SQL)) {

            statement.setLong(1, saleId);
            statement.setInt(2, lineNumber);
            statement.setLong(3, item.productId());
            statement.setInt(4, item.quantity());
            statement.setBigDecimal(5, unitPrice);

            statement.executeUpdate();
        }
    }

    /**
     * Отменяет транзакцию.
     * Если откат также завершился ошибкой, она добавляется
     * к исходному исключению.
     */
    private void rollback(
            Connection connection,
            Throwable originalException) {

        try {
            connection.rollback();
        } catch (SQLException rollbackException) {
            originalException.addSuppressed(rollbackException);
        }
    }

    private List<Sale> executeListQuery(
            String sql,
            Long parameter,
            String operation) {

        Map<Long, Sale> sales = new LinkedHashMap<>();

        try (Connection connection = databaseManager.getConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {

            if (parameter != null) {
                statement.setLong(1, parameter);
            }

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    Long saleId = resultSet.getLong("sale_id");

                    Sale sale = sales.computeIfAbsent(
                            saleId,
                            id -> mapSale(resultSet)
                    );

                    SaleItem item = mapSaleItem(resultSet);

                    if (item != null) {
                        sale.getItems().add(item);
                    }
                }
            }

            return new ArrayList<>(sales.values());

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось " + operation,
                    exception
            );
        }
    }

    private Sale mapSale(ResultSet resultSet) {
        try {
            Customer customer = new Customer(
                    resultSet.getLong("customer_id"),
                    resultSet.getString("full_name"),
                    resultSet.getString("phone"),
                    resultSet.getString("email"),
                    resultSet.getString("document_reference")
            );

            Role role = new Role(
                    resultSet.getLong("role_id"),
                    RoleType.valueOf(resultSet.getString("role_name")),
                    resultSet.getString("role_description")
            );

            User user = new User(
                    resultSet.getLong("user_id"),
                    resultSet.getString("login"),
                    role,
                    resultSet.getBoolean("is_active"),
                    resultSet.getTimestamp("created_at").toLocalDateTime()
            );

            return new Sale(
                    resultSet.getLong("sale_id"),
                    customer,
                    user,
                    resultSet.getTimestamp("sale_date").toLocalDateTime(),
                    SaleStatus.valueOf(resultSet.getString("status")),
                    PaymentType.valueOf(resultSet.getString("payment_type")),
                    new ArrayList<>()
            );

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось преобразовать данные продажи",
                    exception
            );
        }
    }

    private SaleItem mapSaleItem(ResultSet resultSet)
            throws SQLException {

        long productId = resultSet.getLong("product_id");

        if (resultSet.wasNull()) {
            return null;
        }

        ProductType productType = ProductType.valueOf(
                resultSet.getString("product_type")
        );

        Category category = new Category(
                resultSet.getLong("category_id"),
                resultSet.getString("category_name"),
                resultSet.getString("category_description"),
                productType
        );

        Product product = productFactory.createProduct(
                productId,
                category,
                resultSet.getString("manufacturer"),
                resultSet.getString("model"),
                resultSet.getString("product_name"),
                resultSet.getString("specification"),
                resultSet.getBigDecimal("price"),
                resultSet.getInt("stock_quantity"),
                resultSet.getBoolean("product_is_active")
        );

        return new SaleItem(
                resultSet.getInt("line_number"),
                product,
                resultSet.getInt("quantity"),
                resultSet.getBigDecimal("unit_price")
        );
    }

    private void validateId(Long id, String entityName) {
        if (id == null || id <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор " + entityName
                            + " должен быть положительным"
            );
        }
    }
    /**
     * Блокирует строку продажи до завершения транзакции.
     *
     * Используется для предотвращения параллельной регистрации
     * платежей по одной продаже.
     *
     * @param connection соединение текущей транзакции
     * @param saleId идентификатор продажи
     */
    @Override
    public void lockById(Connection connection, Long saleId) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        validateId(saleId, "продажи");

        String sql = """
            SELECT sale_id
            FROM sales
            WHERE sale_id = ?
            FOR UPDATE
            """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setLong(1, saleId);

            try (ResultSet resultSet = statement.executeQuery()) {

                if (!resultSet.next()) {
                    throw new ValidationException(
                            "Продажа с идентификатором "
                                    + saleId + " не найдена"
                    );
                }
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось заблокировать продажу "
                            + saleId,
                    exception
            );
        }
    }
    /**
     * Изменяет статус продажи в рамках существующей транзакции.
     *
     * Используется при регистрации платежей, чтобы изменение
     * статуса продажи выполнялось атомарно с сохранением платежа.
     *
     * @param connection соединение текущей транзакции
     * @param saleId идентификатор продажи
     * @param status новый статус продажи
     */
    @Override
    public void updateStatus(
            Connection connection,
            Long saleId,
            SaleStatus status
    ) {

        if (connection == null) {
            throw new IllegalArgumentException(
                    "Соединение с базой данных не должно быть null"
            );
        }

        validateId(saleId, "продажи");

        if (status == null) {
            throw new IllegalArgumentException(
                    "Статус продажи не должен быть null"
            );
        }

        String sql = """
            UPDATE sales
            SET status = ?
            WHERE sale_id = ?
            """;

        try (PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setString(1, status.name());
            statement.setLong(2, saleId);

            int rowsUpdated = statement.executeUpdate();

            if (rowsUpdated == 0) {
                throw new ValidationException(
                        "Продажа с идентификатором "
                                + saleId + " не найдена"
                );
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось изменить статус продажи "
                            + saleId,
                    exception
            );
        }
    }

    /**
     * Регистрирует платёж с проверкой полномочий сотрудника.
     *
     * @param payment регистрируемый платёж
     * @param operator сотрудник, регистрирующий платёж
     * @return зарегистрированный платёж с идентификатором
     */
    public Payment registerPayment(
            Payment payment,
            User operator
    ) {
        validateOperator(operator);
        validatePayment(payment);

        BigDecimal amount = normalizeAmount(payment.getAmount());

        payment.setAmount(amount);

        return databaseManager.executeInTransaction(connection -> {

            PaymentPurpose purpose = payment.getPaymentPurpose();

            if (purpose == PaymentPurpose.FULL_PAYMENT) {
                return registerFullPayment(connection, payment);
            }

            return registerInstallmentPayment(connection, payment);
        });

    /**
     * Проверяет сотрудника, выполняющего операцию.
     */
    private void validateOperator(User operator) {

        if (operator == null
                || operator.getUserId() == null
                || operator.getUserId() <= 0) {
            throw new ValidationException(
                    "Необходимо указать сотрудника"
            );
        }

        if (!operator.isActive()) {
            throw new BusinessRuleException(
                    "Неактивный сотрудник не может регистрировать платежи"
            );
        }

        if (operator.getRole() == null
                || operator.getRole().getRoleType() == null) {
            throw new BusinessRuleException(
                    "У сотрудника не определена роль"
            );
        }

        RoleType roleType = operator.getRole().getRoleType();

        if (roleType != RoleType.ADMIN
                && roleType != RoleType.SALES_MANAGER) {
            throw new BusinessRuleException(
                    "Недостаточно прав для регистрации платежей"
            );
        }
    }

    /**
     * Проверяет общие данные платежа.
     */
    private void validatePayment(Payment payment) {

        if (payment == null) {
            throw new ValidationException(
                    "Данные платежа не должны быть null"
            );
        }

        if (payment.getPaymentId() != null) {
            throw new ValidationException(
                    "Новый платёж не должен иметь идентификатор"
            );
        }

        if (payment.getSale() == null
                || payment.getSale().getSaleId() == null
                || payment.getSale().getSaleId() <= 0) {
            throw new ValidationException(
                    "Необходимо указать продажу"
            );
        }

        if (payment.getAmount() == null
                || payment.getAmount().signum() <= 0) {
            throw new ValidationException(
                    "Сумма платежа должна быть больше нуля"
            );
        }

        if (payment.getPaymentMethod() == null) {
            throw new ValidationException(
                    "Необходимо указать способ оплаты"
            );
        }

        if (payment.getPaymentPurpose() == null) {
            throw new ValidationException(
                    "Необходимо указать назначение платежа"
            );
        }

        if (payment.getPaymentDate() == null) {
            throw new ValidationException(
                    "Необходимо указать дату платежа"
            );
        }

        if (payment.getPaymentDate().isAfter(LocalDateTime.now())) {
            throw new ValidationException(
                    "Дата платежа не может быть в будущем"
            );
        }

        if (payment.getPaymentPurpose()
                == PaymentPurpose.INSTALLMENT_PAYMENT) {

            if (payment.getApplication() == null
                    || payment.getApplication().getApplicationId() == null) {
                throw new ValidationException(
                        "Для платежа по графику необходимо указать заявку"
                );
            }

            if (payment.getPaymentNumber() == null
                    || payment.getPaymentNumber() <= 0) {
                throw new ValidationException(
                        "Необходимо указать номер платежа по графику"
                );
            }
        } else {
            if (payment.getApplication() != null
                    || payment.getPaymentNumber() != null) {
                throw new ValidationException(
                        "Для данного назначения платежа "
                                + "заявка и номер платежа не указываются"
                );
            }
        }
    }

    /**
     * Приводит сумму к точности денежного значения.
     * Значения с более чем двумя значащими десятичными знаками
     * отклоняются, а не округляются.
     */
    private BigDecimal normalizeAmount(BigDecimal amount) {

        try {
            return amount.setScale(2, RoundingMode.UNNECESSARY);
        } catch (ArithmeticException exception) {
            throw new ValidationException(
                    "Сумма платежа должна содержать не более двух "
                            + "знаков после запятой"
            );
        }
    }

    /**
     * Регистрирует оплату обычной продажи.
     */
    private Payment registerFullPayment(
            java.sql.Connection connection,
            Payment payment
    ) {

        Long saleId = payment.getSale().getSaleId();

        // Блокировка продажи предотвращает параллельную переплату.
        saleDao.lockById(connection, saleId);

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new ValidationException(
                        "Продажа не найдена"
                ));

        validateSaleForPayment(sale);

        if (sale.getPaymentType() != PaymentType.FULL_PAYMENT) {
            throw new BusinessRuleException(
                    "Полная оплата допускается только "
                            + "для продажи с типом FULL_PAYMENT"
            );
        }

        if (sale.getStatus() != SaleStatus.REGISTERED) {
            throw new BusinessRuleException(
                    "Оплата возможна только зарегистрированной продажи"
            );
        }

        List<Payment> existingPayments =
                paymentDao.findBySaleId(saleId);

        BigDecimal alreadyPaid = sumPayments(
                existingPayments,
                PaymentPurpose.FULL_PAYMENT,
                null
        );

        BigDecimal totalAmount = normalizeAmount(
                sale.getTotalAmount()
        );

        BigDecimal remaining = totalAmount.subtract(alreadyPaid);

        if (remaining.signum() <= 0) {
            throw new BusinessRuleException(
                    "Продажа уже полностью оплачена"
            );
        }

        if (payment.getAmount().compareTo(remaining) > 0) {
            throw new BusinessRuleException(
                    "Сумма платежа превышает остаток задолженности. "
                            + "Остаток: " + remaining.toPlainString()
            );
        }

        payment.setSale(sale);

        Long paymentId = paymentDao.create(connection, payment);
        payment.setPaymentId(paymentId);

        BigDecimal projectedPaid =
                alreadyPaid.add(payment.getAmount());

        if (projectedPaid.compareTo(totalAmount) == 0) {
            saleDao.updateStatus(
                    connection,
                    saleId,
                    SaleStatus.PAID
            );
        }

        return payment;
    }

    /**
     * Регистрирует первоначальный взнос
     * или очередной платёж по рассрочке.
     */
    private Payment registerInstallmentPayment(
            java.sql.Connection connection,
            Payment payment
    ) {

        Long applicationId =
                payment.getApplication().getApplicationId();

        // Сериализуем все платежи по одной заявке.
        applicationDao.lockById(connection, applicationId);

        InstallmentApplication application =
                applicationDao.findById(applicationId)
                        .orElseThrow(() -> new ValidationException(
                                "Заявка на рассрочку не найдена"
                        ));

        Sale sale = application.getSale();

        if (sale == null || sale.getSaleId() == null) {
            throw new ValidationException(
                    "У заявки отсутствует продажа"
            );
        }

        sale = saleDao.findById(sale.getSaleId())
                .orElseThrow(() -> new ValidationException(
                        "Продажа заявки не найдена"
                ));

        validateSaleForPayment(sale);

        if (sale.getPaymentType() != PaymentType.INSTALLMENT) {
            throw new BusinessRuleException(
                    "Платежи по рассрочке возможны только "
                            + "для продажи типа INSTALLMENT"
            );
        }

        if (sale.getStatus() != SaleStatus.REGISTERED) {
            throw new BusinessRuleException(
                    "Платежи возможны только по зарегистрированной продаже"
            );
        }

        if (application.getStatus()
                == InstallmentApplicationStatus.COMPLETED) {
            throw new BusinessRuleException(
                    "Рассрочка уже полностью погашена"
            );
        }

        if (payment.getPaymentPurpose()
                == PaymentPurpose.DOWN_PAYMENT) {

            return registerDownPayment(
                    connection,
                    payment,
                    application,
                    sale
            );
        }

        return registerSchedulePayment(
                connection,
                payment,
                application,
                sale
        );
    }

    /**
     * Проверяет состояние продажи.
     */
    private void validateSaleForPayment(Sale sale) {

        if (sale == null) {
            throw new ValidationException(
                    "Продажа не найдена"
            );
        }

        if (sale.getStatus() == SaleStatus.COMPLETED) {
            throw new BusinessRuleException(
                    "Нельзя регистрировать платеж по закрытой продаже"
            );
        }

        if (sale.getStatus() == SaleStatus.CANCELLED) {
            throw new BusinessRuleException(
                    "Нельзя регистрировать платеж по отменённой продаже"
            );
        }

        if (sale.getStatus() == SaleStatus.PAID) {
            throw new BusinessRuleException(
                    "Продажа уже полностью оплачена"
            );
        }
    }

    /**
     * Регистрирует первоначальный взнос.
     * Допускается внесение взноса частями.
     */
    private Payment registerDownPayment(
            java.sql.Connection connection,
            Payment payment,
            InstallmentApplication application,
            Sale sale
    ) {

        if (application.getStatus()
                != InstallmentApplicationStatus.APPROVED) {
            throw new BusinessRuleException(
                    "Первоначальный взнос можно вносить "
                            + "только по одобренной заявке"
            );
        }

        List<Payment> existingPayments =
                paymentDao.findByApplicationId(
                        application.getApplicationId()
                );

        BigDecimal alreadyPaid = sumPayments(
                existingPayments,
                PaymentPurpose.DOWN_PAYMENT,
                null
        );

        BigDecimal requiredAmount = normalizeAmount(
                application.getDownPaymentAmount()
        );

        BigDecimal remaining =
                requiredAmount.subtract(alreadyPaid);

        if (remaining.signum() <= 0) {
            throw new BusinessRuleException(
                    "Первоначальный взнос уже полностью оплачен"
            );
        }

        if (payment.getAmount().compareTo(remaining) > 0) {
            throw new BusinessRuleException(
                    "Сумма платежа превышает остаток первоначального взноса. "
                            + "Остаток: " + remaining.toPlainString()
            );
        }

        payment.setSale(sale);

        Long paymentId = paymentDao.create(connection, payment);
        payment.setPaymentId(paymentId);

        // После первоначального взноса статус остаётся APPROVED.
        // IN_PROGRESS устанавливается при первом платеже по графику.

        return payment;
    }

    /**
     * Регистрирует очередной платёж по графику рассрочки.
     */
    private Payment registerSchedulePayment(
            java.sql.Connection connection,
            Payment payment,
            InstallmentApplication application,
            Sale sale
    ) {

        InstallmentApplicationStatus applicationStatus =
                application.getStatus();

        if (applicationStatus
                != InstallmentApplicationStatus.APPROVED
                && applicationStatus
                != InstallmentApplicationStatus.IN_PROGRESS) {
            throw new BusinessRuleException(
                    "Платежи по графику возможны только "
                            + "по одобренной или действующей рассрочке"
            );
        }

        List<Payment> existingPayments =
                paymentDao.findByApplicationId(
                        application.getApplicationId()
                );

        // Первоначальный взнос должен быть полностью оплачен.
        BigDecimal downPaymentPaid = sumPayments(
                existingPayments,
                PaymentPurpose.DOWN_PAYMENT,
                null
        );

        BigDecimal requiredDownPayment = normalizeAmount(
                application.getDownPaymentAmount()
        );

        if (downPaymentPaid.compareTo(requiredDownPayment) < 0) {
            throw new BusinessRuleException(
                    "Нельзя оплачивать график до полного внесения "
                            + "первоначального взноса. Остаток: "
                            + requiredDownPayment
                            .subtract(downPaymentPaid)
                            .toPlainString()
            );
        }

        List<PaymentScheduleItem> schedule =
                paymentScheduleDao.findByApplicationId(
                        application.getApplicationId()
                );

        if (schedule == null || schedule.isEmpty()) {
            throw new BusinessRuleException(
                    "График платежей не сформирован"
            );
        }

        schedule = schedule.stream()
                .sorted(Comparator.comparingInt(
                        PaymentScheduleItem::getPaymentNumber
                ))
                .toList();

        Map<Integer, BigDecimal> paidByNumber =
                calculatePaidByScheduleNumber(existingPayments);

        PaymentScheduleItem firstUnpaid = null;

        for (PaymentScheduleItem item : schedule) {

            BigDecimal paid = paidByNumber.getOrDefault(
                    item.getPaymentNumber(),
                    ZERO
            );

            BigDecimal planned = normalizeAmount(
                    item.getPlannedAmount()
            );

            if (paid.compareTo(planned) < 0) {
                firstUnpaid = item;
                break;
            }
        }

        if (firstUnpaid == null) {
            throw new BusinessRuleException(
                    "Все платежи по графику уже оплачены"
            );
        }

        if (!Objects.equals(
                payment.getPaymentNumber(),
                firstUnpaid.getPaymentNumber()
        )) {
            throw new BusinessRuleException(
                    "Нельзя пропускать платежи по графику. "
                            + "Следующий платеж: №"
                            + firstUnpaid.getPaymentNumber()
            );
        }

        BigDecimal plannedAmount = normalizeAmount(
                firstUnpaid.getPlannedAmount()
        );

        BigDecimal alreadyPaid = paidByNumber.getOrDefault(
                firstUnpaid.getPaymentNumber(),
                ZERO
        );

        BigDecimal remaining =
                plannedAmount.subtract(alreadyPaid);

        if (payment.getAmount().compareTo(remaining) > 0) {
            throw new BusinessRuleException(
                    "Сумма платежа превышает остаток платежа №"
                            + firstUnpaid.getPaymentNumber()
                            + ". Остаток: "
                            + remaining.toPlainString()
            );
        }

        payment.setSale(sale);
        payment.setApplication(application);

        Long paymentId = paymentDao.create(connection, payment);
        payment.setPaymentId(paymentId);

        BigDecimal projectedPaidForCurrent =
                alreadyPaid.add(payment.getAmount());

        paidByNumber.put(
                firstUnpaid.getPaymentNumber(),
                projectedPaidForCurrent
        );

        // Начало погашения графика переводит заявку
        // в статус IN_PROGRESS.
        if (applicationStatus
                == InstallmentApplicationStatus.APPROVED) {
            application.setStatus(
                    InstallmentApplicationStatus.IN_PROGRESS
            );
        }

        boolean scheduleCompleted = schedule.stream()
                .allMatch(item -> paidByNumber.getOrDefault(
                        item.getPaymentNumber(),
                        ZERO
                ).compareTo(normalizeAmount(item.getPlannedAmount())) >= 0);

        if (scheduleCompleted) {

            application.setStatus(
                    InstallmentApplicationStatus.COMPLETED
            );

            applicationDao.update(
                    connection,
                    application
            );

            saleDao.updateStatus(
                    connection,
                    sale.getSaleId(),
                    SaleStatus.PAID
            );

        } else if (applicationStatus
                != InstallmentApplicationStatus.IN_PROGRESS) {

            applicationDao.update(
                    connection,
                    application
            );
        }

        return payment;
    }

    /**
     * Рассчитывает суммы оплаченных платежей
     * по каждому номеру графика.
     */
    private Map<Integer, BigDecimal> calculatePaidByScheduleNumber(
            List<Payment> payments
    ) {

        Map<Integer, BigDecimal> result = new HashMap<>();

        if (payments == null) {
            return result;
        }

        for (Payment payment : payments) {

            if (payment == null
                    || payment.getPaymentPurpose()
                    != PaymentPurpose.INSTALLMENT_PAYMENT
                    || payment.getPaymentNumber() == null
                    || payment.getAmount() == null) {
                continue;
            }

            int paymentNumber = payment.getPaymentNumber();

            BigDecimal current = result.getOrDefault(
                    paymentNumber,
                    ZERO
            );

            result.put(
                    paymentNumber,
                    current.add(payment.getAmount())
            );
        }

        return result;
    }

    /**
     * Суммирует платежи с заданным назначением.
     *
     * При необходимости учитывает номер платежа по графику.
     */
    private BigDecimal sumPayments(
            List<Payment> payments,
            PaymentPurpose purpose,
            Integer paymentNumber
    ) {

        BigDecimal result = ZERO;

        if (payments == null) {
            return result;
        }

        for (Payment payment : payments) {

            if (payment == null
                    || payment.getAmount() == null
                    || payment.getPaymentPurpose() != purpose) {
                continue;
            }

            if (paymentNumber != null
                    && !Objects.equals(
                    payment.getPaymentNumber(),
                    paymentNumber
            )) {
                continue;
            }

            result = result.add(payment.getAmount());
        }

        return result;
    }
}

-- Схема БД системы учёта продаж компьютерной техники с рассрочкой.
-- СУБД: PostgreSQL 18 (скрипт совместим с PostgreSQL 14 и выше).

--   sales.status               REGISTERED – оформлена, PAID – оплачена, COMPLETED – завершена
--   sales.payment_type         FULL_PAYMENT – полная оплата, INSTALLMENT – рассрочка
--   installment_applications.status
--       CREATED – создана, ON_REVIEW – на рассмотрении, APPROVED – одобрена,
--       REJECTED – отклонена, RETURNED – возвращена на уточнение,
--       IN_PROGRESS – выполняется, COMPLETED – завершена
--   payments.payment_method    CASH – наличные, CARD – карта, TRANSFER – перевод
--   payments.payment_purpose   FULL_PAYMENT – полная оплата, DOWN_PAYMENT – первоначальный взнос,

CREATE TABLE customers (

    CONSTRAINT chk_customer_phone CHECK (btrim(phone) <> '')
);

-- 4. Категории товаров (тип товара определяется категорией)
CREATE TABLE categories (
    category_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL UNIQUE,
    description   VARCHAR(255),
    product_type  VARCHAR(30)  NOT NULL,

    CONSTRAINT chk_category_product_type
        CHECK (product_type IN ('LAPTOP', 'DESKTOP_COMPUTER', 'MONITOR', 'PERIPHERAL_DEVICE'))
);

-- 5. Товары
CREATE TABLE products (
    product_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    category_id    BIGINT        NOT NULL REFERENCES categories (category_id),
    manufacturer   VARCHAR(100)  NOT NULL,
    model          VARCHAR(100)  NOT NULL,
    product_name   VARCHAR(200)  NOT NULL,
    specification  TEXT,
    price          DECIMAL(12, 2) NOT NULL,
    stock_quantity INTEGER       NOT NULL DEFAULT 0,
    is_active      BOOLEAN       NOT NULL DEFAULT TRUE,

    CONSTRAINT chk_product_price CHECK (price >= 0),
    CONSTRAINT chk_product_stock CHECK (stock_quantity >= 0)
);

-- 6. Продажи
CREATE TABLE sales (
    sale_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    customer_id  BIGINT      NOT NULL REFERENCES customers (customer_id),
    user_id      BIGINT      NOT NULL REFERENCES users (user_id),
    sale_date    TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status       VARCHAR(30) NOT NULL DEFAULT 'REGISTERED',
    payment_type VARCHAR(30) NOT NULL,

    CONSTRAINT chk_sale_status
        CHECK (status IN ('REGISTERED', 'PAID', 'COMPLETED')),
    CONSTRAINT chk_sale_payment_type
        CHECK (payment_type IN ('FULL_PAYMENT', 'INSTALLMENT'))
);

-- 7. Состав продаж (зависимая таблица, идентифицирующая связь с sales)
CREATE TABLE sale_items (
    sale_id     BIGINT         NOT NULL REFERENCES sales (sale_id),
    line_number INTEGER        NOT NULL,
    product_id  BIGINT         NOT NULL REFERENCES products (product_id),
    quantity    INTEGER        NOT NULL,
    unit_price  DECIMAL(12, 2) NOT NULL,

    PRIMARY KEY (sale_id, line_number),
    CONSTRAINT uq_sale_product       UNIQUE (sale_id, product_id),
    CONSTRAINT chk_sale_item_line    CHECK (line_number > 0),
    CONSTRAINT chk_sale_item_qty     CHECK (quantity > 0),
    CONSTRAINT chk_sale_item_price   CHECK (unit_price >= 0)
);

-- 8. Заявки на рассрочку
--    Сумма продажи, сумма рассрочки и статус платежей вычисляются и не хранятся.
CREATE TABLE installment_applications (
    application_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    sale_id             BIGINT         NOT NULL UNIQUE REFERENCES sales (sale_id),
    reviewer_id         BIGINT         REFERENCES users (user_id),
    application_date    TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    decision_date       TIMESTAMP,
    term_months         INTEGER        NOT NULL,
    down_payment_amount DECIMAL(12, 2) NOT NULL DEFAULT 0,
    interest_rate       DECIMAL(5, 2)  NOT NULL DEFAULT 0,
    status              VARCHAR(30)    NOT NULL DEFAULT 'CREATED',
    decision_reason     VARCHAR(255),

    CONSTRAINT chk_app_term     CHECK (term_months > 0),
    CONSTRAINT chk_app_down     CHECK (down_payment_amount >= 0),
    CONSTRAINT chk_app_rate     CHECK (interest_rate >= 0),
    CONSTRAINT chk_app_status
        CHECK (status IN ('CREATED', 'ON_REVIEW', 'APPROVED', 'REJECTED',
                          'RETURNED', 'IN_PROGRESS', 'COMPLETED')),
    CONSTRAINT chk_app_decision_date
        CHECK (decision_date IS NULL OR decision_date >= application_date),
    -- отказ и возврат на уточнение всегда сопровождаются основанием
    CONSTRAINT chk_app_reason
        CHECK (status NOT IN ('REJECTED', 'RETURNED')
               OR (decision_reason IS NOT NULL AND btrim(decision_reason) <> '')),
    -- нужен для составного внешнего ключа из payments (платёж и заявка относятся к одной продаже)
    CONSTRAINT uq_app_sale_pair UNIQUE (sale_id, application_id)
);

-- 9. График платежей (зависимая таблица, идентифицирующая связь с заявкой)
CREATE TABLE payment_schedule (
    application_id BIGINT         NOT NULL REFERENCES installment_applications (application_id),
    payment_number INTEGER        NOT NULL,
    due_date       DATE           NOT NULL,
    planned_amount DECIMAL(12, 2) NOT NULL,

    PRIMARY KEY (application_id, payment_number),
    CONSTRAINT chk_schedule_number CHECK (payment_number > 0),
    CONSTRAINT chk_schedule_amount CHECK (planned_amount > 0)
);

-- 10. Платежи
--     Платёж по графику ссылается на строку графика; платежи за полную оплату
--     и первоначальный взнос на график не ссылаются (application_id и payment_number = NULL).
CREATE TABLE payments (
    payment_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    sale_id          BIGINT         NOT NULL REFERENCES sales (sale_id),
    application_id   BIGINT,
    payment_number   INTEGER,
    payment_date     TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    amount           DECIMAL(12, 2) NOT NULL,
    payment_method   VARCHAR(30)    NOT NULL,
    payment_purpose  VARCHAR(30)    NOT NULL,
    reference_number VARCHAR(100),

    CONSTRAINT chk_payment_amount  CHECK (amount > 0),
    CONSTRAINT chk_payment_method  CHECK (payment_method IN ('CASH', 'CARD', 'TRANSFER')),
    CONSTRAINT chk_payment_purpose
        CHECK (payment_purpose IN ('FULL_PAYMENT', 'DOWN_PAYMENT', 'INSTALLMENT_PAYMENT')),

    -- ключи графика заполняются либо оба, либо ни один
    CONSTRAINT chk_payment_schedule_pair
        CHECK ((application_id IS NULL) = (payment_number IS NULL)),
    -- ссылка на график обязательна для платежа по графику и запрещена для остальных
    CONSTRAINT chk_payment_purpose_schedule
        CHECK ((payment_purpose = 'INSTALLMENT_PAYMENT') = (application_id IS NOT NULL)),

    CONSTRAINT fk_payment_schedule
        FOREIGN KEY (application_id, payment_number)
        REFERENCES payment_schedule (application_id, payment_number),
    -- заявка принадлежит той же продаже, что и платёж
    CONSTRAINT fk_payment_sale_application
        FOREIGN KEY (sale_id, application_id)
        REFERENCES installment_applications (sale_id, application_id)
);

-- 11. Журнал действий
CREATE TABLE audit_log (
    log_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id     BIGINT       REFERENCES users (user_id),
    event_time  TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    action      VARCHAR(100) NOT NULL,
    entity_type VARCHAR(100) NOT NULL,
    entity_id   BIGINT,
    details     TEXT
);

-- Индексы по внешним ключам и частым фильтрам
CREATE INDEX idx_users_role          ON users (role_id);
CREATE INDEX idx_products_category   ON products (category_id);

CREATE INDEX idx_sales_user          ON sales (user_id);
CREATE INDEX idx_sales_date          ON sales (sale_date);

CREATE INDEX idx_payments_schedule   ON payments (application_id, payment_number);
CREATE INDEX idx_audit_time          ON audit_log (event_time);
CREATE INDEX idx_audit_user          ON audit_log (user_id);
