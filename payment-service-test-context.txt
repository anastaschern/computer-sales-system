package by.bsuir.computersales.common.model;

import java.io.Serializable;
import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.Objects;

/**
 * Модель заявки на оформление рассрочки.
 */
public class InstallmentApplication implements Serializable {

    private static final long serialVersionUID = 1L;

    private Long applicationId;
    private Sale sale;
    private User reviewer;
    private LocalDateTime applicationDate;
    private LocalDateTime decisionDate;
    private int termMonths;
    private BigDecimal downPaymentAmount;
    private BigDecimal interestRate;
    private InstallmentApplicationStatus status;
    private String decisionReason;

    public InstallmentApplication() {
        this.applicationDate = LocalDateTime.now();
        this.downPaymentAmount = BigDecimal.ZERO;
        this.interestRate = BigDecimal.ZERO;
        this.status = InstallmentApplicationStatus.CREATED;
    }

    public InstallmentApplication(
            Long applicationId,
            Sale sale,
            User reviewer,
            LocalDateTime applicationDate,
            LocalDateTime decisionDate,
            int termMonths,
            BigDecimal downPaymentAmount,
            BigDecimal interestRate,
            InstallmentApplicationStatus status,
            String decisionReason) {

        setApplicationId(applicationId);
        setSale(sale);
        setReviewer(reviewer);
        setApplicationDate(applicationDate);
        setDecisionDate(decisionDate);
        setTermMonths(termMonths);
        setDownPaymentAmount(downPaymentAmount);
        setInterestRate(interestRate);
        setStatus(status);
        setDecisionReason(decisionReason);
    }

    public Long getApplicationId() {
        return applicationId;
    }

    public void setApplicationId(Long applicationId) {
        if (applicationId != null && applicationId <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор заявки должен быть положительным"
            );
        }
        this.applicationId = applicationId;
    }

    public Sale getSale() {
        return sale;
    }

    public void setSale(Sale sale) {
        this.sale = Objects.requireNonNull(
                sale, "Продажа не должна быть null"
        );
    }

    public User getReviewer() {
        return reviewer;
    }

    public void setReviewer(User reviewer) {
        this.reviewer = reviewer;
    }

    public LocalDateTime getApplicationDate() {
        return applicationDate;
    }

    public void setApplicationDate(LocalDateTime applicationDate) {
        this.applicationDate = Objects.requireNonNull(
                applicationDate, "Дата заявки не должна быть null"
        );

        if (decisionDate != null && decisionDate.isBefore(applicationDate)) {
            throw new IllegalArgumentException(
                    "Дата решения не может быть раньше даты заявки"
            );
        }
    }

    public LocalDateTime getDecisionDate() {
        return decisionDate;
    }

    public void setDecisionDate(LocalDateTime decisionDate) {
        if (decisionDate != null
                && applicationDate != null
                && decisionDate.isBefore(applicationDate)) {
            throw new IllegalArgumentException(
                    "Дата решения не может быть раньше даты заявки"
            );
        }
        this.decisionDate = decisionDate;
    }

    public int getTermMonths() {
        return termMonths;
    }

    public void setTermMonths(int termMonths) {
        if (termMonths <= 0) {
            throw new IllegalArgumentException(
                    "Срок рассрочки должен быть больше нуля"
            );
        }
        this.termMonths = termMonths;
    }

    public BigDecimal getDownPaymentAmount() {
        return downPaymentAmount;
    }

    public void setDownPaymentAmount(BigDecimal downPaymentAmount) {
        if (downPaymentAmount == null || downPaymentAmount.signum() < 0) {
            throw new IllegalArgumentException(
                    "Первоначальный взнос не может быть null или отрицательным"
            );
        }
        this.downPaymentAmount = downPaymentAmount;
    }

    public BigDecimal getInterestRate() {
        return interestRate;
    }

    public void setInterestRate(BigDecimal interestRate) {
        if (interestRate == null || interestRate.signum() < 0) {
            throw new IllegalArgumentException(
                    "Процентная ставка не может быть null или отрицательной"
            );
        }
        this.interestRate = interestRate;
    }

    public InstallmentApplicationStatus getStatus() {
        return status;
    }

    public void setStatus(InstallmentApplicationStatus status) {
        this.status = Objects.requireNonNull(
                status, "Статус заявки не должен быть null"
        );
    }

    public String getDecisionReason() {
        return decisionReason;
    }

    public void setDecisionReason(String decisionReason) {
        if ((status == InstallmentApplicationStatus.REJECTED
                || status == InstallmentApplicationStatus.RETURNED)
                && (decisionReason == null || decisionReason.isBlank())) {
            throw new IllegalArgumentException(
                    "Для отклонённой или возвращённой заявки необходимо указать причину"
            );
        }
        this.decisionReason = decisionReason;
    }
}

     * Проверяет согласованность назначения платежа
     * и наличия ссылки на заявку и номер платежа.
     */
    public void validatePaymentReferences() {
        boolean hasApplication = application != null;
        boolean hasPaymentNumber = paymentNumber != null;

        if (hasApplication != hasPaymentNumber) {
            throw new IllegalArgumentException(
                    "Заявка и номер платежа должны быть указаны одновременно"
            );
        }

        boolean installmentPayment =
                paymentPurpose == PaymentPurpose.INSTALLMENT_PAYMENT;

        if (installmentPayment != hasApplication) {
            throw new IllegalArgumentException(
                    "Для платежа по рассрочке необходимо указать заявку "
                            + "и номер платежа; для остальных назначений "
                            + "эти поля должны отсутствовать"
            );
        }

        if (application != null
                && application.getSale() != null
                && sale != null
                && !Objects.equals(
                application.getSale().getSaleId(),
                sale.getSaleId())) {
            throw new IllegalArgumentException(
                    "Заявка должна относиться к той же продаже, что и платёж"
            );
        }
    }
}

package by.bsuir.computersales.common.model;

import java.io.Serializable;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

/**
 * Модель продажи компьютерной техники.
 */
public class Sale implements Serializable {

    private static final long serialVersionUID = 1L;

    private Long saleId;
    private Customer customer;
    private User user;
    private LocalDateTime saleDate;
    private SaleStatus status;
    private PaymentType paymentType;
    private List<SaleItem> items;

    public Sale() {
        this.items = new ArrayList<>();
    }

    public Sale(
            Long saleId,
            Customer customer,
            User user,
            LocalDateTime saleDate,
            SaleStatus status,
            PaymentType paymentType,
            List<SaleItem> items) {

        this();

        this.saleId = saleId;
        setCustomer(customer);
        setUser(user);
        setSaleDate(saleDate);
        setStatus(status);
        setPaymentType(paymentType);
        setItems(items);
    }

    public Long getSaleId() {
        return saleId;
    }

    public void setSaleId(Long saleId) {
        if (saleId != null && saleId <= 0) {
            throw new IllegalArgumentException(
                    "Идентификатор продажи должен быть положительным"
            );
        }

        this.saleId = saleId;
    }

    public Customer getCustomer() {
        return customer;
    }

    public void setCustomer(Customer customer) {
        this.customer = Objects.requireNonNull(
                customer,
                "Покупатель не должен быть null"
        );
    }

    public User getUser() {
        return user;
    }

    public void setUser(User user) {
        this.user = Objects.requireNonNull(
                user,
                "Сотрудник не должен быть null"
        );
    }

    public LocalDateTime getSaleDate() {
        return saleDate;
    }

    public void setSaleDate(LocalDateTime saleDate) {
        this.saleDate = Objects.requireNonNull(
                saleDate,
                "Дата продажи не должна быть null"
        );
    }

    public SaleStatus getStatus() {
        return status;
    }

    public void setStatus(SaleStatus status) {
        this.status = Objects.requireNonNull(
                status,
                "Статус продажи не должен быть null"
        );
    }

    public PaymentType getPaymentType() {
        return paymentType;
    }

    public void setPaymentType(PaymentType paymentType) {
        this.paymentType = Objects.requireNonNull(
                paymentType,
                "Тип оплаты не должен быть null"
        );
    }

    public List<SaleItem> getItems() {
        return items;
    }

    public void setItems(List<SaleItem> items) {
        Objects.requireNonNull(
                items,
                "Список товаров не должен быть null"
        );

        if (items.stream().anyMatch(Objects::isNull)) {
            throw new IllegalArgumentException(
                    "Список не должен содержать null"
            );
        }

        this.items = new ArrayList<>(items);
    }

    /**
     * Возвращает общую стоимость продажи.
     */
    public java.math.BigDecimal getTotalAmount() {
        return items.stream()
                .map(SaleItem::getTotalPrice)
                .reduce(
                        java.math.BigDecimal.ZERO,
                        java.math.BigDecimal::add
                );
    }
}

                properties, "server.thread-pool-size"
        );

        this.sessionTimeoutSeconds = getIntProperty(
                properties, "server.session-timeout-seconds"
        );

        this.databaseUrl = getRequiredProperty(
                properties, "database.url"
        );

        this.databaseUsername = getRequiredProperty(
                properties, "database.username"
        );

        this.databasePassword = getRequiredProperty(
                properties, "database.password"
        );

        this.databasePoolSize = getIntProperty(
                properties, "database.pool-size"
        );

        this.minInstallmentTerm = getIntProperty(
                properties, "rules.term.min"
        );

        this.maxInstallmentTerm = getIntProperty(
                properties, "rules.term.max"
        );

        this.minDownPaymentPercent = getBigDecimalProperty(
                properties, "rules.down_payment.min_percent"
        );

        this.maxFinancedAmount = getBigDecimalProperty(
                properties, "rules.financed.max"
        );

        this.maxActiveInstallments = getIntProperty(
                properties, "rules.active.max"
        );

        validate();
    }

    /**
     * Загружает настройки из указанного файла.
     *
     * @param path путь к файлу настроек
     * @return объект конфигурации сервера
     */
    public static ServerConfig load(Path path) {
        if (path == null) {
            throw new IllegalArgumentException(
                    "Путь к файлу настроек не может быть null"
            );
        }

        Properties properties = new Properties();

        try (InputStream input = Files.newInputStream(path)) {
            properties.load(input);
        } catch (IOException exception) {
            throw new IllegalStateException(
                    "Не удалось загрузить настройки сервера",
                    exception
            );
        }

        return new ServerConfig(properties);
    }

    /**
     * Получает обязательное строковое свойство.
     */
    private static String getRequiredProperty(
            Properties properties,
            String key
    ) {
        String value = properties.getProperty(key);

        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "Не задан обязательный параметр: " + key
            );
        }

        return value.trim();
    }

    /**
     * Получает целочисленное свойство.
     */
    private static int getIntProperty(
            Properties properties,
            String key
    ) {
        String value = getRequiredProperty(properties, key);

        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException exception) {
            throw new IllegalArgumentException(
                    "Параметр " + key + " должен быть целым числом",
                    exception
            );
        }
    }

     */
    private static BigDecimal getBigDecimalProperty(

            String key
    ) {
        String value = getRequiredProperty(properties, key);

        try {
            return new BigDecimal(value);
        } catch (NumberFormatException exception) {
            throw new IllegalArgumentException(
                    "Параметр " + key + " должен быть числом",

            );
        }
    }

package by.bsuir.computersales.server.database;

import by.bsuir.computersales.server.config.ServerConfig;
import by.bsuir.computersales.server.exception.DataAccessException;
import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;

import java.sql.Connection;
import java.sql.SQLException;

/**
 * Управляет пулом подключений к базе данных
 * и транзакциями приложения.
 */
public class DatabaseManager implements AutoCloseable {

    private final HikariDataSource dataSource;

    /**
     * Операция, выполняемая внутри транзакции.
     *
     * @param <T> тип результата
     */
    @FunctionalInterface
    public interface TransactionCallback<T> {

        T execute(Connection connection) throws SQLException;
    }

    /**
     * Создаёт пул подключений на основании настроек сервера.
     *
     * @param serverConfig настройки сервера и базы данных
     */
    public DatabaseManager(ServerConfig serverConfig) {
        if (serverConfig == null) {
            throw new IllegalArgumentException(
                    "Настройки сервера не должны быть null"
            );
        }

        HikariConfig config = new HikariConfig();

        config.setJdbcUrl(serverConfig.getDatabaseUrl());
        config.setUsername(serverConfig.getDatabaseUsername());
        config.setPassword(serverConfig.getDatabasePassword());

        config.setMaximumPoolSize(serverConfig.getDatabasePoolSize());
        config.setPoolName("ComputerSalesPool");
        config.setConnectionTimeout(30000);

        this.dataSource = new HikariDataSource(config);
    }

    /**
     * Возвращает соединение с базой данных.
     *
     * @return соединение JDBC
     * @throws SQLException если получить соединение не удалось
     */
    public Connection getConnection() throws SQLException {
        return dataSource.getConnection();
    }

    /**
     * Выполняет операцию в одной транзакции.
     *
     * Если операция завершается успешно, изменения фиксируются.
     * Если возникает ошибка, все изменения отменяются.
     *
     * @param callback операция, выполняемая в транзакции
     * @param <T> тип результата
     * @return результат операции
     */
    public <T> T executeInTransaction(
            TransactionCallback<T> callback) {

        if (callback == null) {
            throw new IllegalArgumentException(
                    "Операция транзакции не должна быть null"
            );
        }

        try (Connection connection = getConnection()) {

            connection.setAutoCommit(false);

            try {
                T result = callback.execute(connection);

                connection.commit();

                return result;

            } catch (SQLException | RuntimeException exception) {

                try {
                    connection.rollback();
                } catch (SQLException rollbackException) {
                    exception.addSuppressed(rollbackException);
                }

                if (exception instanceof SQLException) {
                    throw new DataAccessException(
                            "Ошибка выполнения транзакции",
                            exception
                    );
                }

                throw (RuntimeException) exception;
            }

        } catch (SQLException exception) {
            throw new DataAccessException(
                    "Не удалось выполнить транзакцию с базой данных",
                    exception
            );
        }
    }

    /**
     * Закрывает пул подключений.
     */
    @Override
    public void close() {
        if (dataSource != null && !dataSource.isClosed()) {
            dataSource.close();
        }
    }
}

        );
    }

    /**
     * Регистрирует платёж с проверкой полномочий сотрудника.
     *
     * @param payment регистрируемый платёж
     * @param operator сотрудник, регистрирующий платёж
     * @return зарегистрированный платёж с идентификатором
     */
    public Payment registerPayment(
            Payment payment,
            User operator
    ) {
        validateOperator(operator);
        validatePayment(payment);

        BigDecimal amount = normalizeAmount(payment.getAmount());

        payment.setAmount(amount);

        return databaseManager.executeInTransaction(connection -> {

            PaymentPurpose purpose = payment.getPaymentPurpose();

            if (purpose == PaymentPurpose.FULL_PAYMENT) {
                return registerFullPayment(connection, payment);
            }

            return registerInstallmentPayment(connection, payment);
        });
    }

    /**
     * Проверяет сотрудника, выполняющего операцию.
     */
    private void validateOperator(User operator) {

        if (operator == null
                || operator.getUserId() == null
                || operator.getUserId() <= 0) {
            throw new ValidationException(
                    "Необходимо указать сотрудника"
            );
        }

        if (!operator.isActive()) {
            throw new BusinessRuleException(
                    "Неактивный сотрудник не может регистрировать платежи"
            );
        }

        if (operator.getRole() == null
                || operator.getRole().getRoleType() == null) {
            throw new BusinessRuleException(
                    "У сотрудника не определена роль"
            );
        }

        RoleType roleType = operator.getRole().getRoleType();

        if (roleType != RoleType.ADMIN
                && roleType != RoleType.SALES_MANAGER) {
            throw new BusinessRuleException(
                    "Недостаточно прав для регистрации платежей"
            );
        }
    }

    /**
     * Проверяет общие данные платежа.
     */
    private void validatePayment(Payment payment) {

        if (payment == null) {
            throw new ValidationException(
                    "Данные платежа не должны быть null"
            );
        }

        if (payment.getPaymentId() != null) {
            throw new ValidationException(
                    "Новый платёж не должен иметь идентификатор"
            );
        }

        if (payment.getSale() == null
                || payment.getSale().getSaleId() == null
                || payment.getSale().getSaleId() <= 0) {
            throw new ValidationException(
                    "Необходимо указать продажу"
            );
        }

        if (payment.getAmount() == null
                || payment.getAmount().signum() <= 0) {
            throw new ValidationException(
                    "Сумма платежа должна быть больше нуля"
            );
        }

        if (payment.getPaymentMethod() == null) {
            throw new ValidationException(
                    "Необходимо указать способ оплаты"
            );
        }

        if (payment.getPaymentPurpose() == null) {
            throw new ValidationException(
                    "Необходимо указать назначение платежа"
            );
        }

        if (payment.getPaymentDate() == null) {
            throw new ValidationException(
                    "Необходимо указать дату платежа"
            );
        }

        if (payment.getPaymentDate().isAfter(LocalDateTime.now())) {
            throw new ValidationException(
                    "Дата платежа не может быть в будущем"
            );
        }

        if (payment.getPaymentPurpose()
                == PaymentPurpose.INSTALLMENT_PAYMENT) {

            if (payment.getApplication() == null
                    || payment.getApplication().getApplicationId() == null) {
                throw new ValidationException(
                        "Для платежа по графику необходимо указать заявку"
                );
            }

            if (payment.getPaymentNumber() == null
                    || payment.getPaymentNumber() <= 0) {
                throw new ValidationException(
                        "Необходимо указать номер платежа по графику"
                );
            }
        } else {
            if (payment.getApplication() != null
                    || payment.getPaymentNumber() != null) {
                throw new ValidationException(
                        "Для данного назначения платежа "
                                + "заявка и номер платежа не указываются"
                );
            }
        }
    }

    /**
     * Приводит сумму к точности денежного значения.
     * Значения с более чем двумя значащими десятичными знаками
     * отклоняются, а не округляются.
     */
    private BigDecimal normalizeAmount(BigDecimal amount) {

        try {
            return amount.setScale(2, RoundingMode.UNNECESSARY);
        } catch (ArithmeticException exception) {
            throw new ValidationException(
                    "Сумма платежа должна содержать не более двух "
                            + "знаков после запятой"
            );
        }
    }

    /**
     * Регистрирует оплату обычной продажи.
     */
    private Payment registerFullPayment(
            java.sql.Connection connection,
            Payment payment
    ) {

        Long saleId = payment.getSale().getSaleId();

        // Блокировка продажи предотвращает параллельную переплату.
        saleDao.lockById(connection, saleId);

        Sale sale = saleDao.findById(saleId)
                .orElseThrow(() -> new ValidationException(
                        "Продажа не найдена"
                ));

        validateSaleForPayment(sale);

        if (sale.getPaymentType() != PaymentType.FULL_PAYMENT) {
            throw new BusinessRuleException(
                    "Полная оплата допускается только "
                            + "для продажи с типом FULL_PAYMENT"
            );
        }

        if (sale.getStatus() != SaleStatus.REGISTERED) {
            throw new BusinessRuleException(
                    "Оплата возможна только зарегистрированной продажи"
            );
        }

        List<Payment> existingPayments =
                paymentDao.findBySaleId(saleId);

        BigDecimal alreadyPaid = sumPayments(
                existingPayments,
                PaymentPurpose.FULL_PAYMENT,
                null
        );

        BigDecimal totalAmount = normalizeAmount(
                sale.getTotalAmount()
        );

        BigDecimal remaining = totalAmount.subtract(alreadyPaid);

        if (remaining.signum() <= 0) {
            throw new BusinessRuleException(
                    "Продажа уже полностью оплачена"
            );
        }

        if (payment.getAmount().compareTo(remaining) > 0) {
            throw new BusinessRuleException(
                    "Сумма платежа превышает остаток задолженности. "
                            + "Остаток: " + remaining.toPlainString()
            );
        }

        payment.setSale(sale);

        Long paymentId = paymentDao.create(connection, payment);
        payment.setPaymentId(paymentId);

        BigDecimal projectedPaid =
                alreadyPaid.add(payment.getAmount());

        if (projectedPaid.compareTo(totalAmount) == 0) {
            saleDao.updateStatus(
                    connection,
                    saleId,
                    SaleStatus.PAID
            );
        }

        return payment;
    }

    /**
     * Регистрирует первоначальный взнос
     * или очередной платёж по рассрочке.
     */
    private Payment registerInstallmentPayment(
            java.sql.Connection connection,
            Payment payment
    ) {

        Long applicationId =
                payment.getApplication().getApplicationId();

        // Сериализуем все платежи по одной заявке.
        applicationDao.lockById(connection, applicationId);

        InstallmentApplication application =
                applicationDao.findById(applicationId)
                        .orElseThrow(() -> new ValidationException(
                                "Заявка на рассрочку не найдена"
                        ));

        Sale sale = application.getSale();

        if (sale == null || sale.getSaleId() == null) {
            throw new ValidationException(
                    "У заявки отсутствует продажа"
            );
        }

        sale = saleDao.findById(sale.getSaleId())
                .orElseThrow(() -> new ValidationException(
                        "Продажа заявки не найдена"
                ));

        validateSaleForPayment(sale);

        if (sale.getPaymentType() != PaymentType.INSTALLMENT) {
            throw new BusinessRuleException(
                    "Платежи по рассрочке возможны только "
                            + "для продажи типа INSTALLMENT"
            );
        }

        if (sale.getStatus() != SaleStatus.REGISTERED) {
            throw new BusinessRuleException(
                    "Платежи возможны только по зарегистрированной продаже"
            );
        }

        if (application.getStatus()
                == InstallmentApplicationStatus.COMPLETED) {
            throw new BusinessRuleException(
                    "Рассрочка уже полностью погашена"
            );
        }

        if (payment.getPaymentPurpose()
                == PaymentPurpose.DOWN_PAYMENT) {

            return registerDownPayment(
                    connection,
                    payment,
                    application,
                    sale
            );
        }

        return registerSchedulePayment(
                connection,
                payment,
                application,
                sale
        );
    }

    /**
     * Проверяет состояние продажи.
     */
    private void validateSaleForPayment(Sale sale) {

        if (sale == null) {
            throw new ValidationException(
                    "Продажа не найдена"
            );
        }

        if (sale.getStatus() == SaleStatus.COMPLETED) {
            throw new BusinessRuleException(
                    "Нельзя регистрировать платеж по закрытой продаже"
            );
        }

        if (sale.getStatus() == SaleStatus.CANCELLED) {
            throw new BusinessRuleException(
                    "Нельзя регистрировать платеж по отменённой продаже"
            );
        }

        if (sale.getStatus() == SaleStatus.PAID) {
            throw new BusinessRuleException(
                    "Продажа уже полностью оплачена"
            );
        }
    }

    /**
     * Регистрирует первоначальный взнос.
     * Допускается внесение взноса частями.
     */
    private Payment registerDownPayment(
            java.sql.Connection connection,
            Payment payment,
            InstallmentApplication application,
            Sale sale
    ) {

        if (application.getStatus()
                != InstallmentApplicationStatus.APPROVED) {
            throw new BusinessRuleException(
                    "Первоначальный взнос можно вносить "
                            + "только по одобренной заявке"
            );
        }

        List<Payment> existingPayments =
                paymentDao.findByApplicationId(
                        application.getApplicationId()
                );

        BigDecimal alreadyPaid = sumPayments(
                existingPayments,
                PaymentPurpose.DOWN_PAYMENT,
                null
        );

        BigDecimal requiredAmount = normalizeAmount(
                application.getDownPaymentAmount()
        );

        BigDecimal remaining =
                requiredAmount.subtract(alreadyPaid);

        if (remaining.signum() <= 0) {
            throw new BusinessRuleException(
                    "Первоначальный взнос уже полностью оплачен"
            );
        }

        if (payment.getAmount().compareTo(remaining) > 0) {
            throw new BusinessRuleException(
                    "Сумма платежа превышает остаток первоначального взноса. "
                            + "Остаток: " + remaining.toPlainString()
            );
        }

        payment.setSale(sale);

        Long paymentId = paymentDao.create(connection, payment);
        payment.setPaymentId(paymentId);

        // После первоначального взноса статус остаётся APPROVED.
        // IN_PROGRESS устанавливается при первом платеже по графику.

        return payment;
    }

    /**
     * Регистрирует очередной платёж по графику рассрочки.
     */
    private Payment registerSchedulePayment(
            java.sql.Connection connection,
            Payment payment,
            InstallmentApplication application,
            Sale sale
    ) {

        InstallmentApplicationStatus applicationStatus =
                application.getStatus();

        if (applicationStatus
                != InstallmentApplicationStatus.APPROVED
                && applicationStatus
                != InstallmentApplicationStatus.IN_PROGRESS) {
            throw new BusinessRuleException(
                    "Платежи по графику возможны только "
                            + "по одобренной или действующей рассрочке"
            );
        }

        List<Payment> existingPayments =
                paymentDao.findByApplicationId(
                        application.getApplicationId()
                );

        // Первоначальный взнос должен быть полностью оплачен.
        BigDecimal downPaymentPaid = sumPayments(
                existingPayments,
                PaymentPurpose.DOWN_PAYMENT,
                null
        );

        BigDecimal requiredDownPayment = normalizeAmount(
                application.getDownPaymentAmount()
        );

        if (downPaymentPaid.compareTo(requiredDownPayment) < 0) {
            throw new BusinessRuleException(
                    "Нельзя оплачивать график до полного внесения "
                            + "первоначального взноса. Остаток: "
                            + requiredDownPayment
                            .subtract(downPaymentPaid)
                            .toPlainString()
            );
        }

        List<PaymentScheduleItem> schedule =
                paymentScheduleDao.findByApplicationId(
                        application.getApplicationId()
                );

        if (schedule == null || schedule.isEmpty()) {
            throw new BusinessRuleException(
                    "График платежей не сформирован"
            );
        }

        schedule = schedule.stream()
                .sorted(Comparator.comparingInt(
                        PaymentScheduleItem::getPaymentNumber
                ))
                .toList();

        Map<Integer, BigDecimal> paidByNumber =
                calculatePaidByScheduleNumber(existingPayments);

        PaymentScheduleItem firstUnpaid = null;

        for (PaymentScheduleItem item : schedule) {

            BigDecimal paid = paidByNumber.getOrDefault(
                    item.getPaymentNumber(),
                    ZERO
            );

            BigDecimal planned = normalizeAmount(
                    item.getPlannedAmount()
            );

            if (paid.compareTo(planned) < 0) {
                firstUnpaid = item;
                break;
            }
        }

        if (firstUnpaid == null) {
            throw new BusinessRuleException(
                    "Все платежи по графику уже оплачены"
            );
        }

        if (!Objects.equals(
                payment.getPaymentNumber(),
                firstUnpaid.getPaymentNumber()
        )) {
            throw new BusinessRuleException(
                    "Нельзя пропускать платежи по графику. "
                            + "Следующий платеж: №"
                            + firstUnpaid.getPaymentNumber()
            );
        }

        BigDecimal plannedAmount = normalizeAmount(
                firstUnpaid.getPlannedAmount()
        );

        BigDecimal alreadyPaid = paidByNumber.getOrDefault(
                firstUnpaid.getPaymentNumber(),
                ZERO
        );

        BigDecimal remaining =
                plannedAmount.subtract(alreadyPaid);

        if (payment.getAmount().compareTo(remaining) > 0) {
            throw new BusinessRuleException(
                    "Сумма платежа превышает остаток платежа №"
                            + firstUnpaid.getPaymentNumber()
                            + ". Остаток: "
                            + remaining.toPlainString()
            );
        }

        payment.setSale(sale);
        payment.setApplication(application);

        Long paymentId = paymentDao.create(connection, payment);
        payment.setPaymentId(paymentId);

        BigDecimal projectedPaidForCurrent =
                alreadyPaid.add(payment.getAmount());

        paidByNumber.put(
                firstUnpaid.getPaymentNumber(),
                projectedPaidForCurrent
        );

        // Начало погашения графика переводит заявку
        // в статус IN_PROGRESS.
        if (applicationStatus
                == InstallmentApplicationStatus.APPROVED) {
            application.setStatus(
                    InstallmentApplicationStatus.IN_PROGRESS
            );
        }

        boolean scheduleCompleted = schedule.stream()
                .allMatch(item -> paidByNumber.getOrDefault(
                        item.getPaymentNumber(),
                        ZERO
                ).compareTo(normalizeAmount(item.getPlannedAmount())) >= 0);

        if (scheduleCompleted) {

            application.setStatus(
                    InstallmentApplicationStatus.COMPLETED
            );

            applicationDao.update(
                    connection,
                    application
            );

            saleDao.updateStatus(
                    connection,
                    sale.getSaleId(),
                    SaleStatus.PAID
            );

        } else if (applicationStatus
                != InstallmentApplicationStatus.IN_PROGRESS) {

            applicationDao.update(
                    connection,
                    application
            );
        }

        return payment;
    }

    /**
     * Рассчитывает суммы оплаченных платежей
     * по каждому номеру графика.
     */
    private Map<Integer, BigDecimal> calculatePaidByScheduleNumber(
            List<Payment> payments
    ) {

        Map<Integer, BigDecimal> result = new HashMap<>();

        if (payments == null) {
            return result;
        }

        for (Payment payment : payments) {

            if (payment == null
                    || payment.getPaymentPurpose()
                    != PaymentPurpose.INSTALLMENT_PAYMENT
                    || payment.getPaymentNumber() == null
                    || payment.getAmount() == null) {
                continue;
            }

            int paymentNumber = payment.getPaymentNumber();

            BigDecimal current = result.getOrDefault(
                    paymentNumber,
                    ZERO
            );

            result.put(
                    paymentNumber,
                    current.add(payment.getAmount())
            );
        }

        return result;
    }

    /**
     * Суммирует платежи с заданным назначением.
     *
     * При необходимости учитывает номер платежа по графику.
     */
    private BigDecimal sumPayments(
            List<Payment> payments,
            PaymentPurpose purpose,
            Integer paymentNumber
    ) {

        BigDecimal result = ZERO;

        if (payments == null) {
            return result;
        }

        for (Payment payment : payments) {

            if (payment == null
                    || payment.getAmount() == null
                    || payment.getPaymentPurpose() != purpose) {
                continue;
            }

            if (paymentNumber != null
                    && !Objects.equals(
                    payment.getPaymentNumber(),
                    paymentNumber
            )) {
                continue;
            }

            result = result.add(payment.getAmount());
        }

        return result;
    }
}

        private final Map<Long, List<PaymentScheduleItem>> schedules =
                new HashMap<>();

        void add(
                Long applicationId,
                PaymentScheduleItem item
        ) {
            schedules.computeIfAbsent(
                    applicationId,
                    ignored -> new ArrayList<>()
            ).add(item);
        }

        @Override
        public List<PaymentScheduleItem> findByApplicationId(
                Long applicationId
        ) {
            return schedules.getOrDefault(

                    List.of()
            );

            add(

                    item
            );
        }

        @Override
        public void createAll(List<PaymentScheduleItem> items) {
            for (PaymentScheduleItem item : items) {
                create(item);
            }
        }

        @Override
        public void deleteByApplicationId(Long applicationId) {
            schedules.remove(applicationId);
        }
